# Version: 1.0.0
# Purpose:
# Execute and audit the full RP9/Rx8 validation chain.
# This is FAS 6: orchestration, execution, verification and master reporting.
# It runs FAS 1 to FAS 5 in fixed order when the validator scripts are present.
# It fails if a validator is missing, crashes, reports FAIL, or lacks expected output.
# It writes WHAT/HOW/WHY/RESULT to JSON, Markdown, trace, audit, verify and log.

def now_iso():
    return datetime.now().isoformat(timespec='seconds')


def make_output_dir():
    out_dir = Path.cwd() / OUTPUT_DIR_NAME
    out_dir.mkdir(parents=True, exist_ok=True)
    return out_dir


def write_text(path, content):
    path.write_text(content, encoding='utf-8')


def write_json(path, data):
    path.write_text(json.dumps(data, indent=4, ensure_ascii=False), encoding='utf-8')


def status(pass_value):
    if pass_value:
        return 'PASS'
    return 'FAIL'


def make_test_record(test_id, title, what_is_tested, how_it_is_tested, why_it_is_tested, expected_result, observed_result, passed, data):
    return {
        'test_id': test_id,
        'title': title,
        'vad_som_testas': what_is_tested,
        'hur_det_testas': how_it_is_tested,
        'varfor_det_testas': why_it_is_tested,
        'forvantat_resultat': expected_result,
        'observerat_resultat': observed_result,
        'slutresultat': status(passed),
        'data': data
    }


def read_json_if_exists(path):
    if not path.exists():
        return None
    try:
        return json.loads(path.read_text(encoding='utf-8'))
    except Exception as exc:
        return {
            'READ_ERROR': str(exc)
        }


def extract_global_status(payload):
    if not isinstance(payload, dict):
        return 'MISSING'

    if 'VERIFY' in payload and isinstance(payload['VERIFY'], dict):
        verify = payload['VERIFY']
        if 'global_status' in verify:
            return str(verify['global_status'])
        if 'final_result' in verify:
            return str(verify['final_result'])

    if 'SUMMARY' in payload and isinstance(payload['SUMMARY'], dict):
        summary = payload['SUMMARY']
        if 'global_status' in summary:
            return str(summary['global_status'])
        if 'System_Status' in summary:
            return str(summary['System_Status'])
        if 'final_result' in summary:
            return str(summary['final_result'])

    if 'AUDIT' in payload and isinstance(payload['AUDIT'], dict):
        audit = payload['AUDIT']
        if 'global_status' in audit:
            return str(audit['global_status'])
        if 'final_result' in audit:
            return str(audit['final_result'])

    if 'READ_ERROR' in payload:
        return 'READ_ERROR'

    return 'UNKNOWN'


def test_1_validator_presence(base_dir):
    rows = []
    for item in VALIDATORS:
        path = base_dir / item['script']
        rows.append({
            'phase': item['phase'],
            'script': item['script'],
            'script_path': str(path),
            'exists': path.exists()
        })
    missing = [row for row in rows if not row['exists']]
    passed = len(missing) == 0
    return make_test_record(
        'TEST_1',
        'VALIDATOR_NARVARO',
        'Narvaro av samtliga FAS 1 till FAS 5-validatorer.',
        'Kontrollerar att varje forvantad Python-fil finns i aktuell katalog innan exekvering.',
        'For att masterkedjan inte ska kunna godkannas om nagon validator saknas.',
        'Alla fem validatorfiler ska finnas.',
        'Saknade validatorer: ' + str(len(missing)) + '.',
        passed,
        {
            'validators': rows,
            'missing_validators': missing
        }
    )


def execute_validators(base_dir):
    execution_rows = []
    for item in VALIDATORS:
        script_path = base_dir / item['script']
        if not script_path.exists():
            execution_rows.append({
                'phase': item['phase'],
                'script': item['script'],
                'executed': False,
                'returncode': None,
                'stdout': '',
                'stderr': 'SCRIPT_MISSING',
                'status': 'FAIL'
            })
            continue
        result = subprocess.run(
            [sys.executable, str(script_path)],
            cwd=str(base_dir),
            capture_output=True,
            text=True,
            timeout=120
        )
        execution_rows.append({
            'phase': item['phase'],
            'script': item['script'],
            'executed': True,
            'returncode': result.returncode,
            'stdout': result.stdout,
            'stderr': result.stderr,
            'status': 'PASS' if result.returncode == 0 else 'FAIL'
        })
    return execution_rows


def test_2_validator_execution(execution_rows):
    failed = [row for row in execution_rows if row['status'] != 'PASS']
    passed = len(failed) == 0
    return make_test_record(
        'TEST_2',
        'VALIDATOR_EXEKVERING',
        'Exekvering av FAS 1 till FAS 5 i fast ordning.',
        'Kor varje validator med samma Python-tolk och lagrar returncode, stdout och stderr.',
        'For att masterpipeline ska kontrollera faktisk korning, inte bara filnarvaro.',
        'Alla validatorer ska exekveras med returncode 0.',
        'Misslyckade exekveringar: ' + str(len(failed)) + '.',
        passed,
        {
            'execution_rows': execution_rows,
            'failed_executions': failed
        }
    )


def collect_phase_outputs(base_dir):
    rows = []
    for item in VALIDATORS:
        output_dir = base_dir / item['output_dir']
        verify_path = output_dir / item['verify_file']
        audit_path = output_dir / item['audit_file']
        verify_payload = read_json_if_exists(verify_path)
        audit_payload = read_json_if_exists(audit_path)
        rows.append({
            'phase': item['phase'],
            'script': item['script'],
            'output_dir': str(output_dir),
            'output_dir_exists': output_dir.exists(),
            'verify_file': str(verify_path),
            'verify_file_exists': verify_path.exists(),
            'verify_status': extract_global_status(verify_payload),
            'verify_payload': verify_payload,
            'audit_file': str(audit_path),
            'audit_file_exists': audit_path.exists(),
            'audit_status': extract_global_status(audit_payload),
            'audit_payload': audit_payload
        })
    return rows

        'TEST_3',
        'VERIFY_KONTROLL',

        passed,
        {
            'phase_outputs': phase_outputs,
            'missing_verify_files': missing,

        }
    )

def build_control_log(summary, tests, execution_rows, phase_outputs, out_dir):
    lines = []
    lines.append('RP9 RX8 MASTER VALIDATION PIPELINE CONTROL LOG')
    lines.append('Generated: ' + summary['generated_at'])
    lines.append('Script: ' + summary['script'])
    lines.append('Version: ' + summary['version'])
    lines.append('Output directory: ' + str(out_dir))
    lines.append('Global status: ' + summary['global_status'])
    lines.append('')
    lines.append('EXECUTION ROWS')
    for row in execution_rows:
        lines.append(row['phase'] + ' | ' + row['script'] + ' | returncode=' + str(row['returncode']) + ' | ' + row['status'])
    lines.append('')
    lines.append('PHASE OUTPUTS')
    for row in phase_outputs:
        lines.append(row['phase'] + ' | verify=' + row['verify_status'] + ' | audit=' + row['audit_status'])
    lines.append('')
    lines.append('MASTER TESTS')
    for test in tests:
        lines.append(test['test_id'] + ' | ' + test['slutresultat'] + ' | ' + test['title'])
        lines.append('WHAT: ' + test['vad_som_testas'])
        lines.append('HOW: ' + test['hur_det_testas'])
        lines.append('WHY: ' + test['varfor_det_testas'])
        lines.append('RESULT: ' + test['observerat_resultat'])
        lines.append('')
    return '\n'.join(lines)


def write_master_outputs(out_dir, summary, tests, execution_rows, phase_outputs):
    payload = {
        'SUMMARY': summary,
        'VALIDATORS': VALIDATORS,
        'EXECUTION_ROWS': execution_rows,
        'PHASE_OUTPUTS': phase_outputs,
        'TESTS': tests
    }
    trace_payload = {
        'TRACE': tests,
        'EXECUTION_ROWS': execution_rows
    }
    audit_payload = {
        'AUDIT': {
            'all_tests_include_what_how_why_result': all(
                all(key in test for key in ['vad_som_testas', 'hur_det_testas', 'varfor_det_testas', 'slutresultat'])
                for test in tests
            ),
            'master_pipeline_only': True,
            'no_new_theory_introduced': True,
            'validators': VALIDATORS,
            'global_status': summary['global_status']
        }
    }
    verify_payload = {
        'VERIFY': {
            'test_1_validator_presence': tests[0]['slutresultat'] == 'PASS',
            'test_2_validator_execution': tests[1]['slutresultat'] == 'PASS',
            'test_3_verify_control': tests[2]['slutresultat'] == 'PASS',
            'test_4_audit_control': tests[3]['slutresultat'] == 'PASS',
            'test_5_global_chain_status': tests[4]['slutresultat'] == 'PASS',
            'test_6_master_consistency': tests[5]['slutresultat'] == 'PASS',
            'global_status': summary['global_status']
        }
    }
    write_json(out_dir / 'rp9_rx8_master_data.json', payload)
    write_json(out_dir / 'rp9_rx8_master_trace.json', trace_payload)
    write_json(out_dir / 'rp9_rx8_master_audit.json', audit_payload)
    write_json(out_dir / 'rp9_rx8_master_verify.json', verify_payload)
    markdown = build_markdown(summary, tests, execution_rows, phase_outputs)
    write_text(out_dir / 'rp9_rx8_master_presentation.md', markdown)
    control_log = build_control_log(summary, tests, execution_rows, phase_outputs, out_dir)
    write_text(out_dir / 'rp9_rx8_master_control_log.txt', control_log)


def run_pipeline():
    out_dir = make_output_dir()
    base_dir = Path.cwd()
    tests = []
    presence_test = test_1_validator_presence(base_dir)
    tests.append(presence_test)
    execution_rows = execute_validators(base_dir)
    tests.append(test_2_validator_execution(execution_rows))
    phase_outputs = collect_phase_outputs(base_dir)
    tests.append(test_3_verify_control(phase_outputs))
    tests.append(test_4_audit_control(phase_outputs))
    tests.append(test_5_global_chain_status(phase_outputs))
    tests.append(test_6_master_consistency(tests))
    global_pass_pre_report = all(test['slutresultat'] == 'PASS' for test in tests)
    summary = {
        'system': 'RP9_RX8',
        'phase': 'FAS_6_MASTER_VALIDATION_PIPELINE',
        'script': SCRIPT_NAME,
        'version': SCRIPT_VERSION,
        'generated_at': now_iso(),
        'global_status': status(global_pass_pre_report),
        'principle': 'Master pipeline orchestrates existing validators only.',
        'validation_boundary': 'No new relation, no new geometry, no new residual theory.'
    }
    write_master_outputs(out_dir, summary, tests, execution_rows, phase_outputs)
    tests.append(test_7_master_revision_report(out_dir))
    global_pass = all(test['slutresultat'] == 'PASS' for test in tests)
    summary['global_status'] = status(global_pass)
    write_master_outputs(out_dir, summary, tests, execution_rows, phase_outputs)
    print('=' * 72)
    print('RP9 RX8 MASTER VALIDATION PIPELINE')
    print('STATUS:', summary['global_status'])
    print('OUTPUT:', out_dir)
    print('=' * 72)
    for test in tests:
        print(test['test_id'] + ' | ' + test['slutresultat'] + ' | ' + test['title'])
    print('=' * 72)
    if not global_pass:
        return 1
    return 0


if __name__ == '__main__':
    exit_code = run_pipeline()
    sys.exit(exit_code)
