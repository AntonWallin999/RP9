PROGRAM_NAME = 'SECOND_DIFFERENTIATION_harmonic_probe_V2'

OUT_DIR_NAME = 'SECOND_DIFFERENTIATION_PRESENTATION_V2'


def d(value):
    return Decimal(str(value))


def decimal_fraction(value):
    text = format(float(value), '.15f')
    if '.' not in text:
        return ''
    return text.split('.')[1].rstrip('0')


def repetend_unit_fraction(denominator):
    # What: calculates the repeating cycle for 1 / denominator.
    # How: long division with remainder tracking.
    # Why: K6 needs to test whether 7 acts as a six-step recursive passage.
    seen = {}
    digits = []
    remainder = 1 % denominator
    position = 0
    while remainder and remainder not in seen:
        seen[remainder] = position
        remainder *= 10
        digits.append(str(remainder // denominator))
        remainder %= denominator
        position += 1
    if remainder == 0:
        return ''.join(digits), ''
    start = seen[remainder]
    prefix = ''.join(digits[:start])
    cycle = ''.join(digits[start:])
    return prefix, cycle


def decimal_part_cycle(numerator, denominator):
    # What: calculates the repeating decimal cycle for numerator / denominator.
    # How: long division after extracting the integer part.
    # Why: K6 compares 1/7 and 8/7 to verify preserved six-cycle position.
    seen = {}
    digits = []
    remainder = numerator % denominator
    position = 0
    while remainder and remainder not in seen:
        seen[remainder] = position
        remainder *= 10
        digits.append(str(remainder // denominator))
        remainder %= denominator
        position += 1
    if remainder == 0:
        return ''.join(digits), ''
    start = seen[remainder]
    prefix = ''.join(digits[:start])
    cycle = ''.join(digits[start:])
    return prefix, cycle


def add_test(tests, test_id, title, what, how, why, expected, observed, passed, relation):
    tests.append({
        'test_id': test_id,
        'title': title,
        'what_is_tested': what,
        'how_it_is_tested': how,
        'why_it_is_tested': why,
        'expected_result': expected,
        'observed_result': observed,
        'status': 'PASS' if passed else 'FAIL',
        'relation_to_rp9': relation
    })


def classify_tri_pulse(value):
    frac = decimal_fraction(value)
    if frac.startswith('333') or frac.startswith('333333'):
        return 'ONE_THIRD_PULSE'
    if frac.startswith('666') or frac.startswith('666666'):
        return 'TWO_THIRD_PULSE'
    return 'NO_TRIADIC_PULSE'

def run_probe():
    root = Path(__file__).resolve().parent
    out_dir = root / OUT_DIR_NAME
    out_dir.mkdir(parents=True, exist_ok=True)
    tests = []

    # Measurement layer: first differentiation.
    R = 1.0
    Ri = math.sqrt(3) / 2 * R
    AY = math.pi * (R ** 2)
    AI = math.pi * (Ri ** 2)
    area_ratio = AI / AY
    lock_residual = 1.0 - area_ratio
    area_pass = abs(area_ratio - 0.75) < 2e-16 and abs(lock_residual - 0.25) < 2e-16
    add_test(
        tests,
        'K6_TEST_01',
        'First differentiation lock',
        '3/4 mot 1/4 fran Vesica-baserad forsta differentiering.',
        'Beraknar intern radie sqrt(3)/2, area-kvot AI/AY och komplementet 1 - AI/AY.',
        'Detta ar basen som K6 inte far losgora alpha ifran.',
        'area_ratio ~= 0.75 och lock_residual ~= 0.25',
        {'area_ratio': area_ratio, 'lock_residual': lock_residual},
        area_pass,
        'Alpha far endast placeras efter detta las, inte fore det.'
    )

    # Half-half layer.
    half_high = 1.5 / 2.0
    half_low = 0.5 / 2.0
    half_pass = half_high == 0.75 and half_low == 0.25 and half_high + half_low == 1.0
    add_test(
        tests,
        'K6_TEST_02',
        'Halva-halva lock',
        'Den bipolara halva-halva-funktionen: 1.5/2 och 0.5/2.',
        'Beraknar bada sidorna och kontrollerar att summan stanger till 1.0.',
        'Detta binder K6 till K5-funktionen innan alpha analyseras.',
        '1.5/2 = 0.75, 0.5/2 = 0.25, summa = 1.0',
        {'upper_half': half_high, 'lower_half': half_low, 'sum': half_high + half_low},
        half_pass,
        'Halva-halva ar uttrycket som genererar 3/4 och 1/4 i representationslagret.'
    )

    # Binary triadic pulse from Rx8.
    rx8_values = [6.0, 3.0, 1.5, 0.75, 0.375, 0.1875, 0.09375, 0.046875]
    pulse_rows = []
    for item in rx8_values:
        result = 8.0 / item
        pulse_rows.append({'input': item, 'result': result, 'pulse': classify_tri_pulse(result)})
    pulse_set = [row['pulse'] for row in pulse_rows]
    pulse_pass = all(pulse in ['ONE_THIRD_PULSE', 'TWO_THIRD_PULSE'] for pulse in pulse_set)
    add_test(
        tests,
        'K6_TEST_03',
        'Binar-triadisk pulse',
        'Rx8 dividerat genom halveringskedjan skall reproducera .333... och .666...-vaxlingen.',
        'Dividerar 8 med 6, 3, 1.5, 0.75 och vidare halveringar.',
        'K5 visar att siffrorna uttrycker funktionen, inte tvartom. K6 maste darfor bevara pulsen.',
        'Samtliga resultat skall klassas som ONE_THIRD_PULSE eller TWO_THIRD_PULSE.',
        pulse_rows,
        pulse_pass,
        'Den triadiska pulsen ar K5-bryggan in i K6.'
    )

    # Seven passage.
    _, cycle_1_7 = decimal_part_cycle(1, 7)
    _, cycle_8_7 = decimal_part_cycle(8, 7)
    _, cycle_8_14 = decimal_part_cycle(8, 14)
    seven_pass = cycle_1_7 == '142857' and cycle_8_7 == '142857' and len(cycle_1_7) == 6 and len(cycle_8_14) == 6
    add_test(
        tests,
        'K6_TEST_04',
        'Sju-passagen',
        '1/7 och 8/7 skall bevara samma sexfasiga rekursiva decimalcykel.',
        'Lang division med restspairning for 1/7, 8/7 och 8/14.',
        'Sjuan skall inte skapa ny proportion. Den skall visa passagen som later funktionen rotera.',
        '1/7 och 8/7 ger 142857; cykellangd = 6.',
        {'1_over_7_cycle': cycle_1_7, '8_over_7_cycle': cycle_8_7, '8_over_14_cycle': cycle_8_14},
        seven_pass,
        '7 fungerar som fraktal passage mellan enhet och Rx8-volym.'
    )

    # Alpha layer.
    alpha = 0.0072973525692838015
    alpha_inv_declared = 137.035999084
    alpha_inv_calculated = 1.0 / alpha
    alpha_anchor = 0.007
    alpha_residual = alpha - alpha_anchor
    alpha_relative_to_anchor = alpha / alpha_anchor
    alpha_pass = abs(alpha_inv_calculated - alpha_inv_declared) < 1e-9 and alpha_residual > 0 and alpha_anchor == 7 / 1000
    add_test(
        tests,
        'K6_TEST_05',
        'Finstrukturkonstantens ankare',
        'Alpha testas som finstrukturkonstant med bas-10-ankare 0.007 och positiv asymmetrisk rest.',
        'Kontrollerar 1/alpha mot deklarerad invers, kontrollerar 0.007 = 7/1000 och beraknar alpha - 0.007.',
        'K6 skall inte loskoppla alpha fran sju-passagen; alpha placeras efter den verifierade K5-pulsen.',
        '1/alpha ~= 137.035999084, anchor = 0.007, residual > 0.',
        {'alpha': alpha, 'alpha_inverse_calculated': alpha_inv_calculated, 'alpha_inverse_declared': alpha_inv_declared, 'alpha_anchor': alpha_anchor, 'alpha_residual': alpha_residual, 'alpha_relative_to_anchor': alpha_relative_to_anchor},
        alpha_pass,
        'Alpha binds som harmonisk residual/finstrukturposition efter 7-ankaret, inte som fristaende tal.'
    )

    # Rx8 machine residual lock.
    R16 = 2.0 ** -52
    R15 = 2.0 ** -49
    quotient = R15 / R16
    machine_pass = quotient == 8.0
    add_test(
        tests,
        'K6_TEST_06',
        'Maskinell Rx8-residual',
        'R15/R16 skall lasa till exakt 8.0.',
        'Beraknar 2^-49 / 2^-52.',
        'Detta binder K6 tillbaka till maskinlagret och oktala laspunkten.',
        'R15/R16 = 8.0',
        {'R15': R15, 'R16': R16, 'quotient': quotient},
        machine_pass,
        'Residualen tillhor representationen men reproducerar Rx8-laspunkten.'
    )

    # Combined closure.
    global_pass = all(test['status'] == 'PASS' for test in tests)
    payload = {
        'metadata': {
            'program': PROGRAM_NAME,
            'version': VERSION,
            'timestamp': datetime.datetime.now().isoformat(),
            'license': 'CC BY-SA 4.0 - Anton Wallin',
            'framework': FRAMEWORK
        },
        'rp9_position': {
            'layer': 'K6 SECOND DIFFERENTIATION',
            'function': 'Functional calibration of alpha against K5 half-half pulse, seven passage and Rx8 lock.',
            'not_primary_origin': True,
            'primary_origin': 'R and first differentiation remain upstream.'
        },
        'measurement_layer': {
            'base_radius': R,
            'internal_radius': Ri,
            'area_ratio': area_ratio,
            'lock_residual': lock_residual
        },
        'half_half_layer': {
            'upper_half': half_high,
            'lower_half': half_low,
            'sum': half_high + half_low
        },
        'triadic_pulse_layer': pulse_rows,
        'seven_passage_layer': {
            'one_over_seven_cycle': cycle_1_7,
            'eight_over_seven_cycle': cycle_8_7,
            'eight_over_fourteen_cycle': cycle_8_14
        },
        'alpha_layer': {
            'alpha_value': alpha,
            'alpha_inverse_declared': alpha_inv_declared,
            'alpha_inverse_calculated': alpha_inv_calculated,
            'alpha_base_10_anchor': alpha_anchor,
            'alpha_asymmetric_residual': alpha_residual,
            'alpha_relative_to_anchor': alpha_relative_to_anchor
        },
        'machine_residual_layer': {
            'R15': R15,
            'R16': R16,
            'R15_over_R16': quotient
        },
        'tests': tests,
        'verification': {
            'global_status': 'PASS' if global_pass else 'FAIL',
            'final_result': 'K6 ar funktionellt kalibrerad mot K5 om samtliga testlager passerar: halva-halva, triadisk puls, sju-passage, alpha-ankare och Rx8-maskinresidual.'
        }
    }

    json_path = out_dir / 'SECOND_DIFFERENTIATION_DATA_V2.json'
    md_path = out_dir / 'SECOND_DIFFERENTIATION_PRESENTATION_V2.md'
    log_path = out_dir / 'SECOND_DIFFERENTIATION_CONTROL_LOG_V2.txt'
    py_copy_path = out_dir / 'SECOND_DIFFERENTIATION_harmonic_probe_V2.py'

    json_path.write_text(json.dumps(payload, indent=4, ensure_ascii=False), encoding='utf-8')
    md_path.write_text(build_markdown(payload), encoding='utf-8')
    log_lines = []
    log_lines.append('PROGRAM=' + PROGRAM_NAME)
    log_lines.append('VERSION=' + VERSION)
    log_lines.append('TIMESTAMP=' + payload['metadata']['timestamp'])
    log_lines.append('GLOBAL_STATUS=' + payload['verification']['global_status'])
    for test in tests:
        log_lines.append(test['test_id'] + '=' + test['status'])
    log_path.write_text('\n'.join(log_lines) + '\n', encoding='utf-8')
    Path(__file__).read_text(encoding='utf-8')
    py_copy_path.write_text(Path(__file__).read_text(encoding='utf-8'), encoding='utf-8')

    print('RP9 K6 SECOND DIFFERENTIATION V2')
    print('Output folder:', out_dir)
    print('Global status:', payload['verification']['global_status'])
    for test in tests:
        print(test['test_id'], '|', test['status'], '|', test['title'])
    return payload


if __name__ == '__main__':
    run_probe()
