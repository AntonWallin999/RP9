RP9_CORE4_ORIENTATION_EXTRACTION_IMPLEMENTATION_AUDITOR.py
Version: 1.0.0

Purpose
-------
Audits the remaining implementation question in RP9_CORE4_OPERATOR_STATE_DERIVATION_VALIDATOR.py:

    How is the Core4 orientation extracted?

This script does not attempt to re-declare RP9. It audits whether the previous
validator's orientation extraction is functionally independent of ABC/B5/B6 naming
and whether the extracted operator-state vector can be reproduced by an independent
minimal implementation before any ABC comparison is loaded.

Exit codes
----------
0  = extraction implementation is locked: no forbidden pre-ABC dependency, independent reproduction matches.
10 = extraction implementation is not locked: dependency or reproduction failure.
20 = audit could not run due to missing/invalid inputs.
"""

    "extract_center_pair_axes",

@dataclass
class StaticFunctionAudit:
    function_name: str
    exists: bool
    referenced_names: List[str]
    forbidden_references: List[str]
    pass_status: bool

@dataclass
class RuntimeAudit:
    name: str
    pass_status: bool
    result: str
    details: Dict[str, object]

@dataclass
class IndependentCandidate:
    rotation_deg: Tuple[float, float, float]
    raw_vector: List[float]
    canonical_vector: List[float]
    angle_to_x_deg: float
    angle_to_z_deg: float
    y_abs: float
    passes_rule: bool
    reason: str

# IO -----------------------------------------------------------------------
def ensure_dirs() -> None:
    for p in [OUT_DIR, JSON_DIR, CSV_DIR, REPORT_DIR, AUDIT_DIR]:
        p.mkdir(parents=True, exist_ok=True)

def now() -> str:
    return datetime.now().isoformat(timespec="seconds")

def log(message: str) -> None:
    ensure_dirs()
    line = f"[{now()}] {message}"
    print(line)
    with LOG_PATH.open("a", encoding="utf-8") as h:
        h.write(line + "\n")

def json_safe(obj):
    if isinstance(obj, dict):
        return {str(k): json_safe(v) for k, v in obj.items()}
    if isinstance(obj, list):
        return [json_safe(v) for v in obj]
    if isinstance(obj, tuple):
        return [json_safe(v) for v in obj]
    if isinstance(obj, Path):
        return str(obj)
    if isinstance(obj, np.ndarray):
        return json_safe(obj.tolist())
    if isinstance(obj, np.generic):
        return obj.item()
    if hasattr(obj, "__dataclass_fields__"):
        return json_safe(asdict(obj))
    return obj

def write_json(path: Path, payload: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(json_safe(payload), indent=2, ensure_ascii=False), encoding="utf-8")

def write_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding="utf-8")

def write_csv(path: Path, rows: List[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    if not rows:
        path.write_text("", encoding="utf-8")
        return
    with path.open("w", encoding="utf-8", newline="") as h:
        w = csv.DictWriter(h, fieldnames=list(rows[0].keys()))
        w.writeheader()
        w.writerows(rows)

def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

# Static audit --------------------------------------------------------------
class NameCollector(ast.NodeVisitor):
    def __init__(self) -> None:
        self.names: Set[str] = set()
        self.attrs: Set[str] = set()
    def visit_Name(self, node: ast.Name) -> None:
        self.names.add(node.id)
        self.generic_visit(node)
    def visit_Attribute(self, node: ast.Attribute) -> None:
        self.attrs.add(node.attr)
        self.generic_visit(node)

def parse_target_tree() -> ast.Module:
    if not TARGET_VALIDATOR.exists():
        raise FileNotFoundError(f"Missing target validator: {TARGET_VALIDATOR}")
    return ast.parse(TARGET_VALIDATOR.read_text(encoding="utf-8"))

def find_function(tree: ast.Module, name: str) -> Optional[ast.FunctionDef]:
    for node in tree.body:
        if isinstance(node, ast.FunctionDef) and node.name == name:
            return node
    return None

def audit_static_dependencies(tree: ast.Module) -> List[StaticFunctionAudit]:
    audits: List[StaticFunctionAudit] = []
    for fn_name in PRE_ABC_FUNCTIONS:
        node = find_function(tree, fn_name)
        if node is None:
            audits.append(StaticFunctionAudit(fn_name, False, [], ["FUNCTION_MISSING"], False))
            continue
        collector = NameCollector()
        collector.visit(node)
        refs = sorted(collector.names | collector.attrs)
        forbidden = sorted([x for x in refs if x in FORBIDDEN_PRE_ABC_NAMES or "B5" in x or "B6" in x or "ABC" in x])
        audits.append(StaticFunctionAudit(fn_name, True, refs, forbidden, len(forbidden) == 0))
    return audits

# Independent minimal reproduction -----------------------------------------
def normalize(v: Iterable[float]) -> np.ndarray:
    arr = np.asarray(list(v), dtype=float)
    n = float(np.linalg.norm(arr))
    if n <= EPS:
        raise ValueError("zero-vector")
    return arr / n

def round_vec(v: Iterable[float], decimals: int = 12) -> List[float]:
    arr = np.asarray(list(v), dtype=float)
    arr[np.abs(arr) < 10 ** (-(decimals + 1))] = 0.0
    return [round(float(x), decimals) for x in arr]

def angle_directed(a: Iterable[float], b: Iterable[float]) -> float:
    ua, ub = normalize(a), normalize(b)
    dot = max(-1.0, min(1.0, float(np.dot(ua, ub))))
    return float(math.degrees(math.acos(dot)))

def rotation_matrix(ax_deg: float, ay_deg: float, az_deg: float) -> np.ndarray:
    ax, ay, az = np.deg2rad([ax_deg, ay_deg, az_deg])
    rx = np.array([[1.0,0.0,0.0],[0.0,np.cos(ax),-np.sin(ax)],[0.0,np.sin(ax),np.cos(ax)]])
    ry = np.array([[np.cos(ay),0.0,np.sin(ay)],[0.0,1.0,0.0],[-np.sin(ay),0.0,np.cos(ay)]])
    rz = np.array([[np.cos(az),-np.sin(az),0.0],[np.sin(az),np.cos(az),0.0],[0.0,0.0,1.0]])
    return rz @ ry @ rx

def canonicalize(v: Iterable[float]) -> np.ndarray:
    arr = normalize(v)
    if arr[0] < -EPS or (abs(arr[0]) <= EPS and arr[2] < -EPS):
        arr = -arr
    return arr

def independent_extract(rotations: List[Tuple[float,float,float]], radius: float) -> Tuple[Optional[np.ndarray], List[IndependentCandidate], Dict[str, object]]:
    x = np.array([1.0,0.0,0.0])
    z = np.array([0.0,0.0,1.0])
    base_left = np.array([-radius/2.0,0.0,0.0])
    base_right = np.array([radius/2.0,0.0,0.0])
    candidates: List[IndependentCandidate] = []
    pass_lines: Set[Tuple[float,float,float]] = set()
    for rot in rotations:
        mat = rotation_matrix(*rot)
        left = mat @ base_left
        right = mat @ base_right
        raw = normalize(right - left)
        canon = canonicalize(raw)
        ax = angle_directed(canon, x)
        az = angle_directed(canon, z)
        y_abs = abs(float(canon[1]))
        passes = (y_abs <= EPS and canon[0] > 0 and canon[2] > 0 and abs(ax-45.0) <= ANGLE_TOL_DEG and abs(az-45.0) <= ANGLE_TOL_DEG)
        reason = "passes independent XZ-half-angle / Y-neutral extraction rule" if passes else "does not satisfy independent extraction rule"
        if passes:
            pass_lines.add(tuple(round_vec(canon, 9)))
        candidates.append(IndependentCandidate(tuple(float(v) for v in rot), round_vec(raw), round_vec(canon), ax, az, y_abs, passes, reason))
    if len(pass_lines) == 1:
        v = normalize(list(pass_lines)[0])
        return v, candidates, {"status":"UNIQUE_INDEPENDENT_EXTRACTION", "unique_line_count":1, "passing_candidate_count":sum(c.passes_rule for c in candidates), "derived_vector":round_vec(v)}
    if len(pass_lines) == 0:
        return None, candidates, {"status":"NO_INDEPENDENT_EXTRACTION", "unique_line_count":0, "passing_candidate_count":0, "derived_vector":None}
    return None, candidates, {"status":"MULTIPLE_INDEPENDENT_EXTRACTIONS", "unique_line_count":len(pass_lines), "passing_candidate_count":sum(c.passes_rule for c in candidates), "derived_vectors":[list(x) for x in sorted(pass_lines)]}

# Runtime import ------------------------------------------------------------
def import_target_module():
    spec = importlib.util.spec_from_file_location("target_core4_validator", TARGET_VALIDATOR)
    if spec is None or spec.loader is None:
        raise ImportError("Cannot create spec for target validator")
    module = importlib.util.module_from_spec(spec)
    sys.modules["target_core4_validator"] = module
    spec.loader.exec_module(module)  # type: ignore[attr-defined]
    return module

def compare_vectors(a: Optional[np.ndarray], b: Optional[np.ndarray]) -> Tuple[bool, Optional[float]]:
    if a is None or b is None:
        return False, None
    ang = angle_directed(a, b)
    return ang <= ANGLE_TOL_DEG, ang

def zip_outputs() -> None:
    if ZIP_PATH.exists():
        ZIP_PATH.unlink()
    files = [Path(__file__).resolve()]
    if OUT_DIR.exists():
        files.extend([p for p in OUT_DIR.rglob("*") if p.is_file()])
    with zipfile.ZipFile(ZIP_PATH, "w", compression=zipfile.ZIP_DEFLATED) as zf:
        for p in files:
            zf.write(p, arcname=p.relative_to(ROOT_DIR))

def run() -> int:
    ensure_dirs()
    if LOG_PATH.exists():
        LOG_PATH.unlink()
    log("START CORE4 ORIENTATION EXTRACTION IMPLEMENTATION AUDIT")

    if not TARGET_VALIDATOR.exists():
        log("TARGET_VALIDATOR missing")
        return 20

    tree = parse_target_tree()
    static_audits = audit_static_dependencies(tree)
    static_pass = all(a.pass_status and a.exists for a in static_audits)

    target = import_target_module()
    rotations, rotation_info = target.load_rotations()
    circles = target.build_core4_circles(rotations)
    target_candidates = target.extract_center_pair_axes(circles)
    target_derived, target_state = target.derive_operator_state(target_candidates)

    independent_derived, independent_candidates, independent_state = independent_extract(rotations, float(target.RADIUS))
    vector_match, angle_diff = compare_vectors(target_derived, independent_derived)

    # Taint test: mutate ABC definitions after extraction. This must not alter pre-ABC vector.
    original_fallback = getattr(target, "FALLBACK_ABC_VIEWS", None)
    taint_result = "not_applicable"
    taint_stable = False
    try:
        target.FALLBACK_ABC_VIEWS = {"B5": (-1.0, 0.0, -1.0), "B6": (1.0, 0.0, 1.0), "X_FAKE": (0.0, 1.0, 0.0)}
        taint_circles = target.build_core4_circles(rotations)
        taint_candidates = target.extract_center_pair_axes(taint_circles)
        taint_derived, taint_state = target.derive_operator_state(taint_candidates)
        taint_stable, taint_angle = compare_vectors(target_derived, taint_derived)
        taint_result = f"stable={taint_stable}; angle_diff={taint_angle}; state={taint_state.get('status') if isinstance(taint_state, dict) else taint_state}"
    finally:
        if original_fallback is not None:
            target.FALLBACK_ABC_VIEWS = original_fallback

    source_text = TARGET_VALIDATOR.read_text(encoding="utf-8")
    source_hash = sha256_file(TARGET_VALIDATOR)

    final_pass = bool(static_pass and target_derived is not None and independent_derived is not None and vector_match and taint_stable)
    classification = "ORIENTATION_EXTRACTION_IMPLEMENTATION_LOCKED" if final_pass else "ORIENTATION_EXTRACTION_IMPLEMENTATION_NOT_LOCKED"
    if final_pass:
        statement = (
            "Orienteringsextraktionen är låst på implementeringsnivå: pre-ABC-funktionerna innehåller inga "
            "B5/B6/ABC-referenser, målvalidatorns operator-state-vektor reproduceras av en oberoende minimal "
            "Core4-centerpar-extraktion, och en muterad ABC/B5/B6-referens ändrar inte den pre-ABC-härledda vektorn."
        )
        exit_code = 0
    else:
        statement = (
            "Orienteringsextraktionen är inte slutligt låst: minst en kontroll misslyckades. Granska statisk audit, "
            "oberoende reproduktion eller taint-test innan frågan används deklarativt."
        )
        exit_code = 10

    payload = {
        "metadata": {
            "generated": now(),
            "script": SCRIPT_NAME,
            "version": SCRIPT_VERSION,
            "target_validator": TARGET_VALIDATOR.name,
            "target_sha256": source_hash,
            "rotation_info": rotation_info,
            "angle_tolerance_deg": ANGLE_TOL_DEG,
        },
        "question": "Granska den återstående implementationsdelen som avgör hur Core4-orienteringen extraheras.",
        "static_function_audit": [asdict(a) for a in static_audits],
        "runtime": {
            "target_operator_state": json_safe(target_state),
            "target_operator_vector": round_vec(target_derived) if target_derived is not None else None,
            "independent_operator_state": independent_state,
            "independent_operator_vector": round_vec(independent_derived) if independent_derived is not None else None,
            "target_independent_vector_match": vector_match,
            "target_independent_angle_difference_deg": angle_diff,
            "taint_vector_stable": taint_stable,
            "taint_result": taint_result,
        },
        "independent_extraction_candidates": [asdict(c) for c in independent_candidates],
        "conclusion": {
            "final_pass": final_pass,
            "implementation_lock_status": "LOCKED" if final_pass else "NOT_LOCKED",
            "classification": classification,
            "final_statement": statement,
            "exit_code": exit_code,
        },
        "audit_boundary": {
            "what_this_audit_can_lock": "The implementation path used to extract Core4 orientation before ABC/B5/B6 comparison.",
            "what_this_audit_does_not_retest": "The global uniqueness of B5/B6 among all RP9 observations; that belongs to the separate candidate-elimination validators.",
        },
    }

    write_json(JSON_DIR / "CORE4_ORIENTATION_EXTRACTION_IMPLEMENTATION_AUDIT.json", payload)
    write_csv(CSV_DIR / "STATIC_FUNCTION_AUDIT.csv", [asdict(a) for a in static_audits])
    write_csv(CSV_DIR / "INDEPENDENT_EXTRACTION_CANDIDATES.csv", [asdict(c) for c in independent_candidates])
    write_text(REPORT_DIR / "CORE4_ORIENTATION_EXTRACTION_IMPLEMENTATION_AUDIT.md", build_report(payload))

    audit_hashes = {
        "generated": now(),
        "script": SCRIPT_NAME,
        "version": SCRIPT_VERSION,
        "target_validator": TARGET_VALIDATOR.name,
        "files": {},
    }
    for p in [
        Path(__file__).resolve(), TARGET_VALIDATOR,
        JSON_DIR / "CORE4_ORIENTATION_EXTRACTION_IMPLEMENTATION_AUDIT.json",
        CSV_DIR / "STATIC_FUNCTION_AUDIT.csv",
        CSV_DIR / "INDEPENDENT_EXTRACTION_CANDIDATES.csv",
        REPORT_DIR / "CORE4_ORIENTATION_EXTRACTION_IMPLEMENTATION_AUDIT.md",
        LOG_PATH,
    ]:
        if p.exists():
            audit_hashes["files"][p.name] = {"path": str(p), "sha256": sha256_file(p), "size_bytes": p.stat().st_size}
    write_json(AUDIT_DIR / "CORE4_ORIENTATION_EXTRACTION_IMPLEMENTATION_AUDIT_HASHES.json", audit_hashes)
    zip_outputs()

    log(f"FINAL_PASS={final_pass}")
    log(f"IMPLEMENTATION_LOCK_STATUS={'LOCKED' if final_pass else 'NOT_LOCKED'}")
    log(f"CLASSIFICATION={classification}")
    log(f"STATEMENT={statement}")
    log(f"ZIP={ZIP_PATH}")
    log("END")
    return exit_code

if __name__ == "__main__":
    sys.exit(run())
