RP9. CORE4. (O-S-D-V). PYTHON


RP9_CORE4_OPERATOR_STATE_DERIVATION_VALIDATOR.py
Version: 1.0.0

Purpose
-------
Final structural validator for the remaining Core4 orientation question.

Question tested
---------------
Does Core4 contain an internally derivable operator-state orientation before ABC/B5/B6
classification is loaded, or does the B5/B6 position require a later observation choice?

Method boundary
---------------
This validator does NOT use B5/B6 to derive the operator-state orientation.
It generates Core4 from its circle/rotation structure, extracts relation axes from the
resulting Core4 center-pair geometry, stress-tests alternative orientations, and only
then compares the derived orientation to ABC/B5/B6 definitions.

Output
------
- JSON report
- Markdown report
- CSV tables
- Audit JSON
- Execution log
- ZIP package

Exit codes
----------
0  = H0 rejected; Core4 operator-state orientation is derived and matches ABC after final comparison.
10 = H0 accepted; no unique operator-state orientation exists before ABC comparison.
20 = test definition failed or insufficient source data.

def write_text(path: Path, text: str) -> None:

def write_csv(path: Path, rows: List[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    if not rows:
        path.write_text("", encoding="utf-8")
        return
    with path.open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(rows[0].keys()))
        writer.writeheader()
        writer.writerows(rows)


def json_safe(obj):
    if isinstance(obj, dict):
        return {str(k): json_safe(v) for k, v in obj.items()}
    if isinstance(obj, list):
        return [json_safe(v) for v in obj]
    if isinstance(obj, tuple):
        return [json_safe(v) for v in obj]
    if isinstance(obj, Path):
        return str(obj)
    if isinstance(obj, np.ndarray):
        return json_safe(obj.tolist())
    if isinstance(obj, np.generic):
        return obj.item()
    if hasattr(obj, "__dataclass_fields__"):
        return json_safe(asdict(obj))
    return obj


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def parse_assignment_from_text(text: str, name: str):
    """Extract literal assignment from a Python text fragment."""
    try:
        tree = ast.parse(text)
    except SyntaxError:
        return None
    for node in tree.body:
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name) and target.id == name:
                    try:
                        return ast.literal_eval(node.value)
                    except Exception:
                        return None
    return None


def extract_python_blocks_from_md(path: Path) -> List[str]:
    if not path.exists():
        return []
    text = path.read_text(encoding="utf-8", errors="ignore")
    blocks: List[str] = []
    marker = "````python"
    start = 0
    while True:
        i = text.find(marker, start)
        if i < 0:
            break
        j = text.find("````", i + len(marker))
        if j < 0:
            break
        blocks.append(text[i + len(marker):j].strip())
        start = j + 4
    if not blocks and path.suffix == ".py":
        blocks.append(text)
    return blocks


def load_rotations() -> Tuple[List[Tuple[float, float, float]], Dict[str, str]]:
    candidates = [
        ROOT_DIR / "3.B. Core4X (X-Y-Z). Sequence. PYTHON.md",
        ROOT_DIR / "3.A. Core4X (X-Y-Z). PYTHON.md",
        ROOT_DIR / "Vesica_Render_HTML_Only_AB_Categorized.py",
    ]
    info = {"rotations_source": "fallback"}
    for path in candidates:
        blocks = extract_python_blocks_from_md(path) if path.suffix == ".md" else [path.read_text(encoding="utf-8", errors="ignore")] if path.exists() else []
        for block in blocks:
            rotations = parse_assignment_from_text(block, "ROTATIONS")
            if rotations is None:
                rotations = parse_assignment_from_text(block, "ROTATIONS_GREEN")
            if rotations is not None:
                info["rotations_source"] = path.name
                return [tuple(float(x) for x in rot) for rot in rotations], info
    return FALLBACK_ROTATIONS, info


def load_abc_views() -> Tuple[Dict[str, Tuple[float, float, float]], Dict[str, str]]:
    source = ROOT_DIR / "Vesica_Render_HTML_Only_AB_Categorized.py"
    info = {"abc_source": "fallback"}
    if source.exists():
        text = source.read_text(encoding="utf-8", errors="ignore")
        groups = parse_assignment_from_text(text, "VIEW_GROUPS")
        if isinstance(groups, dict):
            views: Dict[str, Tuple[float, float, float]] = {}
            for group in groups.values():
                if isinstance(group, dict):
                    for name, vec in group.get("views", {}).items():
                        views[str(name)] = tuple(float(x) for x in vec)
            if views:
                info["abc_source"] = source.name
                return views, info
    return FALLBACK_ABC_VIEWS, info

# Geometry primitives

def normalize(v: Iterable[float]) -> np.ndarray:
    arr = np.asarray(list(v), dtype=float)
    n = float(np.linalg.norm(arr))
    if n <= EPS:
        raise ValueError("zero-vector cannot be normalized")
    return arr / n


def round_vec(v: Iterable[float], decimals: int = 12) -> List[float]:
    arr = np.asarray(list(v), dtype=float)
    arr[np.abs(arr) < 10 ** (-(decimals + 1))] = 0.0
    return [round(float(x), decimals) for x in arr]


def angle_directed(a: Iterable[float], b: Iterable[float]) -> float:
    ua, ub = normalize(a), normalize(b)
    dot = max(-1.0, min(1.0, float(np.dot(ua, ub))))
    return float(math.degrees(math.acos(dot)))


def angle_line(a: Iterable[float], b: Iterable[float]) -> float:
    ua, ub = normalize(a), normalize(b)
    dot = max(-1.0, min(1.0, abs(float(np.dot(ua, ub)))))
    return float(math.degrees(math.acos(dot)))


def rotation_matrix(ax_deg: float, ay_deg: float, az_deg: float) -> np.ndarray:
    ax, ay, az = np.deg2rad([ax_deg, ay_deg, az_deg])
    rx = np.array([[1.0, 0.0, 0.0], [0.0, np.cos(ax), -np.sin(ax)], [0.0, np.sin(ax), np.cos(ax)]])
    ry = np.array([[np.cos(ay), 0.0, np.sin(ay)], [0.0, 1.0, 0.0], [-np.sin(ay), 0.0, np.cos(ay)]])
    rz = np.array([[np.cos(az), -np.sin(az), 0.0], [np.sin(az), np.cos(az), 0.0], [0.0, 0.0, 1.0]])
    return rz @ ry @ rx


def canonicalize_for_initial_frame(v: Iterable[float]) -> np.ndarray:
    """Canonical sign for the declared VP-front frame: X must be non-negative; if X=0 use Z."""
    arr = normalize(v)
    if arr[0] < -EPS or (abs(arr[0]) <= EPS and arr[2] < -EPS):
        arr = -arr
    return arr


def line_key(v: Iterable[float]) -> Tuple[float, float, float]:
    arr = canonicalize_for_initial_frame(v)
    return tuple(round_vec(arr, 9))

# Core4 generation and extraction

def build_core4_circles(rotations: List[Tuple[float, float, float]], radius: float = RADIUS) -> List[Core4Circle]:
    base_centers = [np.array([-radius / 2.0, 0.0, 0.0]), np.array([radius / 2.0, 0.0, 0.0])]
    base_ex = np.array([1.0, 0.0, 0.0])
    base_ey = np.array([0.0, 1.0, 0.0])
    base_n = np.array([0.0, 0.0, 1.0])
    circles: List[Core4Circle] = []
    for rot in rotations:
        mat = rotation_matrix(*rot)
        ex = normalize(mat @ base_ex)
        ey = normalize(mat @ base_ey)
        n = normalize(mat @ base_n)
        for idx, center in enumerate(base_centers):
            circles.append(Core4Circle(
                rotation_deg=tuple(float(x) for x in rot),
                pair_index=idx,
                center=round_vec(mat @ center),
                ex=round_vec(ex),
                ey=round_vec(ey),
                normal=round_vec(n),
                radius=radius,
            ))
    return circles


def extract_center_pair_axes(circles: List[Core4Circle]) -> List[AxisCandidate]:
    by_rotation: Dict[Tuple[float, float, float], List[Core4Circle]] = {}
    for circle in circles:
        by_rotation.setdefault(circle.rotation_deg, []).append(circle)
    candidates: List[AxisCandidate] = []
    x = np.array([1.0, 0.0, 0.0])
    z = np.array([0.0, 0.0, 1.0])
    for rot, pair in sorted(by_rotation.items()):
        if len(pair) != 2:
            continue
        pair_sorted = sorted(pair, key=lambda c: c.pair_index)
        c0 = np.asarray(pair_sorted[0].center, dtype=float)
        c1 = np.asarray(pair_sorted[1].center, dtype=float)
        raw = normalize(c1 - c0)
        canon = canonicalize_for_initial_frame(raw)
        angle_x = angle_directed(canon, x)
        angle_z = angle_directed(canon, z)
        y_abs = abs(float(canon[1]))
        dot_x = float(np.dot(canon, x))
        dot_z = float(np.dot(canon, z))
        passes = (
            y_abs <= EPS and
            dot_x > 0.0 and
            dot_z > 0.0 and
            abs(angle_x - 45.0) <= ANGLE_TOL_DEG and
            abs(angle_z - 45.0) <= ANGLE_TOL_DEG
        )
        if passes:
            reason = "candidate is XZ-half-angle, Y-neutral, positive to X and positive to frontal Z"
        else:
            reason = "candidate fails at least one operator-state rule before ABC comparison"
        candidates.append(AxisCandidate(
            source="core4_center_pair_axis",
            rotation_deg=rot,
            raw_vector=round_vec(raw),
            canonical_vector=round_vec(canon),
            angle_to_x_deg=angle_x,
            angle_to_z_deg=angle_z,
            y_abs=y_abs,
            dot_x=dot_x,
            dot_z=dot_z,
            satisfies_operator_state_rule=passes,
            reason=reason,
        ))
    return candidates


def derive_operator_state(candidates: List[AxisCandidate]) -> Tuple[Optional[np.ndarray], Dict[str, object]]:
    passing = [c for c in candidates if c.satisfies_operator_state_rule]
    unique_lines = {tuple(c.canonical_vector) for c in passing}
    rejected = [c for c in candidates if not c.satisfies_operator_state_rule]
    if len(unique_lines) == 1:
        vec = normalize(list(unique_lines)[0])
        return vec, {
            "status": "UNIQUE_OPERATOR_STATE_DERIVED",
            "passing_candidate_count": len(passing),
            "unique_operator_line_count": 1,
            "rejected_candidate_count": len(rejected),
            "derived_vector": round_vec(vec),
        }
    if len(unique_lines) == 0:
        return None, {
            "status": "NO_OPERATOR_STATE_DERIVED",
            "passing_candidate_count": 0,
            "unique_operator_line_count": 0,
            "rejected_candidate_count": len(rejected),
            "derived_vector": None,
        }
    return None, {
        "status": "MULTIPLE_OPERATOR_STATES_DERIVED",
        "passing_candidate_count": len(passing),
        "unique_operator_line_count": len(unique_lines),
        "rejected_candidate_count": len(rejected),
        "derived_vectors": [list(v) for v in sorted(unique_lines)],
    }

def transform_rotations(rotations: List[Tuple[float, float, float]], case: str) -> List[Tuple[float, float, float]]:
    if case == "same_order_control":
        return list(rotations)
    if case == "reversed_order":
        return list(reversed(rotations))
    if case == "shuffled_order_seed_9":
        temp = list(rotations)
        random.Random(9).shuffle(temp)
        return temp
    if case == "invert_x_rotation_signs":
        return [(-x, y, z) for x, y, z in rotations]
    if case == "invert_y_rotation_signs":
        return [(x, -y, z) for x, y, z in rotations]
    if case == "invert_xy_rotation_signs":
        return [(-x, -y, z) for x, y, z in rotations]
    return list(rotations)


def run_falsification(rotations: List[Tuple[float, float, float]], reference_vec: Optional[np.ndarray]) -> List[FalsificationCase]:
    cases = [
        ("same_order_control", "control: same rotations"),
        ("reversed_order", "order falsification: reverse Core4 rotation list"),
        ("shuffled_order_seed_9", "order falsification: deterministic shuffle"),
        ("invert_x_rotation_signs", "orientation stress: invert X-rotation signs"),
        ("invert_y_rotation_signs", "orientation stress: invert Y-rotation signs"),
        ("invert_xy_rotation_signs", "orientation stress: invert X and Y rotation signs"),
    ]
    out: List[FalsificationCase] = []
    for name, desc in cases:
        test_rots = transform_rotations(rotations, name)
        test_circles = build_core4_circles(test_rots)
        test_candidates = extract_center_pair_axes(test_circles)
        derived, state = derive_operator_state(test_candidates)
        alt = False
        result: str
        if derived is None:
            result = f"no unique derived operator state; {state['status']}"
            alt = True
        elif reference_vec is not None and angle_directed(derived, reference_vec) <= ANGLE_TOL_DEG:
            result = "same operator-state orientation retained"
            alt = False
        else:
            result = "different operator-state orientation derived"
            alt = True
        out.append(FalsificationCase(
            case_name=name,
            transformation=desc,
            candidate_count=int(state.get("passing_candidate_count", 0)),
            derived_vector=round_vec(derived) if derived is not None else None,
            h0_supports_alternative=alt,
            result=result,
        ))
    return out

# ABC final comparison
# ---------------------------------------------------------------------------
def compare_to_abc(derived: np.ndarray, abc_views: Dict[str, Tuple[float, float, float]]) -> List[ABCComparison]:
    rows: List[ABCComparison] = []
    for name, vec in abc_views.items():
        v = normalize(vec)
        signed_dot = float(np.dot(derived, v))
        abs_dot = abs(signed_dot)
        d_ang = angle_directed(derived, v)
        l_ang = angle_line(derived, v)
        rows.append(ABCComparison(
            name=name,
            vector=round_vec(v),
            signed_dot=signed_dot,
            abs_dot=abs_dot,
            directed_angle_deg=d_ang,
            line_angle_deg=l_ang,
            directed_match=d_ang <= ANGLE_TOL_DEG,
            line_match=l_ang <= ANGLE_TOL_DEG,
        ))
    rows.sort(key=lambda r: (r.line_angle_deg, r.directed_angle_deg, r.name))
    return rows


def classify_final(operator_state: Dict[str, object], derived: Optional[np.ndarray], falsification: List[FalsificationCase], abc_rows: List[ABCComparison]) -> Dict[str, object]:
    if derived is None:
        return {
            "final_answer_code": "A",
            "h0_status": "ACCEPTED",
            "h1_status": "REJECTED",
            "final_pass": False,
            "exit_code": 10,
            "classification": "CORE4_NO_UNIQUE_OPERATOR_STATE_ORIENTATION",
            "final_statement": "Core4 saknar unik operatoriskt härledd observationsorientering före ABC-jämförelse.",
        }

    alternatives = [case for case in falsification if case.h0_supports_alternative]
    directed = [r.name for r in abc_rows if r.directed_match]
    line = [r.name for r in abc_rows if r.line_match]

    if alternatives:
        return {
            "final_answer_code": "A",
            "h0_status": "ACCEPTED",
            "h1_status": "REJECTED",
            "final_pass": False,
            "exit_code": 10,
            "classification": "H0_SUPPORTED_BY_ALTERNATIVE_OPERATOR_STATE",
            "directed_matches_after_abc": directed,
            "line_matches_after_abc": line,
            "falsification_alternatives": [asdict(a) for a in alternatives],
            "final_statement": "Falsifieringen hittade alternativa operatoriska orienteringar före ABC-jämförelse.",
        }

    if directed == ["B5"] and set(line) == {"B5", "B6"}:
        return {
            "final_answer_code": "B",
            "h0_status": "REJECTED",
            "h1_status": "ACCEPTED",
            "final_pass": True,
            "exit_code": 0,
            "classification": "CORE4_OPERATOR_STATE_DERIVES_B5_WITH_B6_INVERSE_LINE",
            "derived_operator_vector": round_vec(derived),
            "directed_matches_after_abc": directed,
            "line_matches_after_abc": line,
            "final_statement": "Core4 producerar en unik operatoriskt härledd orientering före ABC-jämförelse. Efter ABC-jämförelse sammanfaller riktad position med B5 och axellinjen med B5/B6.",
        }

    if directed == ["B6"] and set(line) == {"B5", "B6"}:
        return {
            "final_answer_code": "C",
            "h0_status": "REJECTED",
            "h1_status": "ACCEPTED",
            "final_pass": True,
            "exit_code": 0,
            "classification": "CORE4_OPERATOR_STATE_DERIVES_B6_WITH_B5_INVERSE_LINE",
            "derived_operator_vector": round_vec(derived),
            "directed_matches_after_abc": directed,
            "line_matches_after_abc": line,
            "final_statement": "Core4 producerar en unik operatoriskt härledd orientering före ABC-jämförelse. Efter ABC-jämförelse sammanfaller riktad position med B6 och axellinjen med B5/B6.",
        }

    return {
        "final_answer_code": "D",
        "h0_status": "REJECTED_AS_NO_ALTERNATIVE_BUT_ABC_MISMATCH",
        "h1_status": "REJECTED_AS_B5B6_MATCH_FAILED",
        "final_pass": False,
        "exit_code": 10,
        "classification": "CORE4_UNIQUE_OPERATOR_STATE_NOT_MATCHING_B5B6",
        "derived_operator_vector": round_vec(derived),
        "directed_matches_after_abc": directed,
        "line_matches_after_abc": line,
        "nearest_abc_view": abc_rows[0].name if abc_rows else None,
        "final_statement": "Core4 producerar en unik operatorisk orientering men den motsvarar inte B5/B6 enligt ABC-jämförelsen.",
    }

def zip_outputs() -> None:
    if ZIP_PATH.exists():
        ZIP_PATH.unlink()
    files = [Path(__file__).resolve()]
    if OUT_DIR.exists():
        files.extend([p for p in OUT_DIR.rglob("*") if p.is_file()])
    with zipfile.ZipFile(ZIP_PATH, "w", compression=zipfile.ZIP_DEFLATED) as zf:
        for p in files:
            zf.write(p, arcname=p.relative_to(ROOT_DIR))


def run() -> int:
    ensure_dirs()
    if LOG_PATH.exists():
        LOG_PATH.unlink()
    log("START CORE4 OPERATOR STATE DERIVATION VALIDATOR")

    references = collect_reference_records()
    rotations, rotation_info = load_rotations()
    circles = build_core4_circles(rotations)
    candidates = extract_center_pair_axes(circles)
    derived, operator_state = derive_operator_state(candidates)
    falsification = run_falsification(rotations, derived)
    abc_views, abc_info = load_abc_views()
    abc_rows = compare_to_abc(derived, abc_views) if derived is not None else []
    conclusion = classify_final(operator_state, derived, falsification, abc_rows)

    self_audit = [
        {
            "question": "Vilka antaganden gjordes?",
            "answer": "Endast startframe används: Vesica Piscis frontläge = XY, X = relationell längd, Z = frontal normal. ABC/B5/B6 används inte i operator-state-deriveringen.",
        },
        {
            "question": "Vilken data användes före ABC?",
            "answer": "Core4-rotationslistan, genererade cirkelcentra och centerpar-axlar.",
        },
        {
            "question": "Vilken data användes inte före ABC?",
            "answer": "ABC-klasser, B5/B6-namn, kandidatunikitetstabeller och tidigare manifest-position-slutsats användes inte för att härleda operator-state-vektorn.",
        },
        {
            "question": "Kan slutsatsen reproduceras utan ABC-definitionerna?",
            "answer": "Ja för operator-state-vektorn. ABC krävs endast i sista jämförelsen för att namnge den som B5/B6-linje.",
        },
        {
            "question": "Vad försökte falsifieringen göra?",
            "answer": "Den försökte stödja H0 genom att skapa flera likvärdiga operator-state-orienteringar före ABC. H0 stöds endast om ett alternativ överlever samma regel.",
        },
    ]

    payload = {
        "metadata": {
            "generated": now(),
            "script": SCRIPT_NAME,
            "version": SCRIPT_VERSION,
            "validator_type": "operator_state_derivation_before_abc_comparison",
            "rotation_info": rotation_info,
            "abc_info": abc_info,
            "angle_tolerance_deg": ANGLE_TOL_DEG,
        },
        "hypothesis_test": {
            "H0": "Core4 saknar unik inneboende observationsorientering före ABC/B5/B6, eller kan orienteras på flera likvärdiga sätt.",
            "H1": "Core4 producerar en unik operator-state-orientering före ABC/B5/B6, och denna kan därefter jämföras mot ABC.",
        },
        "reference_files": [asdict(r) for r in references],
        "generated_core4_circles": [asdict(c) for c in circles],
        "axis_candidates_before_abc": [asdict(c) for c in candidates],
        "operator_state_derivation": operator_state,
        "falsification_cases": [asdict(f) for f in falsification],
        "abc_final_comparison": [asdict(r) for r in abc_rows],
        "conclusion": conclusion,
        "self_audit": self_audit,
    }

    write_json(JSON_DIR / "CORE4_OPERATOR_STATE_DERIVATION_VALIDATION.json", payload)
    write_csv(CSV_DIR / "CORE4_AXIS_CANDIDATES.csv", [asdict(c) for c in candidates])
    write_csv(CSV_DIR / "FALSIFICATION_CASES.csv", [asdict(f) for f in falsification])
    write_csv(CSV_DIR / "ABC_FINAL_COMPARISON.csv", [asdict(r) for r in abc_rows])
    write_text(REPORT_DIR / "CORE4_OPERATOR_STATE_DERIVATION_VALIDATION.md", build_markdown(payload))

    audit_payload = {
        "generated": now(),
        "script": SCRIPT_NAME,
        "version": SCRIPT_VERSION,
        "final_classification": conclusion["classification"],
        "final_pass": conclusion["final_pass"],
        "files": {},
    }
    for p in [JSON_DIR / "CORE4_OPERATOR_STATE_DERIVATION_VALIDATION.json",

              LOG_PATH]:

            audit_payload["files"][p.name] = {"path": str(p), "sha256": sha256_file(p), "size_bytes": p.stat().st_size}
    write_json(AUDIT_DIR / "CORE4_OPERATOR_STATE_DERIVATION_AUDIT.json", audit_payload)

    zip_outputs()
    log(f"FINAL_PASS={conclusion['final_pass']}")
    log(f"FINAL_ANSWER_CODE={conclusion['final_answer_code']}")
    log(f"H0_STATUS={conclusion['h0_status']}")
    log(f"H1_STATUS={conclusion['h1_status']}")
    log(f"CLASSIFICATION={conclusion['classification']}")
    log(f"STATEMENT={conclusion['final_statement']}")
    log(f"ZIP={ZIP_PATH}")
    log("END")
    return int(conclusion["exit_code"])
