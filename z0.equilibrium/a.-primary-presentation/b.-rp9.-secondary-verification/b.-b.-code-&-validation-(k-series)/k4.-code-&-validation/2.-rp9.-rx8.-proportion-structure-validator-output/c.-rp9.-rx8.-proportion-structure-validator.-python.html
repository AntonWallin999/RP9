# -*- coding: utf-8 -*-

# RP9_RX8_PROPORTION_STRUCTURE_VALIDATOR.py
# Version: 1.0.0
# Purpose:
# Validate Phase 2 for RP9/Rx8 as a proportion and structure validator.
# This validator checks whether the same relational proportion is preserved
# through relation, representation, Rx8 capsulation and residual position.
# It does not search for competing external relations.
# It does not import an external reality-test layer.
# It validates proportion through form and structure.

import json
import math
import sys
from datetime import datetime
from fractions import Fraction
from pathlib import Path


SCRIPT_VERSION = '1.0.0'
SCRIPT_NAME = 'RP9_RX8_PROPORTION_STRUCTURE_VALIDATOR.py'
OUTPUT_DIR_NAME = 'RP9_RX8_PROPORTION_STRUCTURE_VALIDATOR_OUTPUT'


def now_iso():
    return datetime.now().isoformat(timespec='seconds')


def make_output_dir():
    out_dir = Path.cwd() / OUTPUT_DIR_NAME
    out_dir.mkdir(parents=True, exist_ok=True)
    return out_dir


def write_text(path, content):
    path.write_text(content, encoding='utf-8')


def write_json(path, data):
    path.write_text(json.dumps(data, indent=4, ensure_ascii=False), encoding='utf-8')


def to_jsonable(value):
    if isinstance(value, Fraction):
        return str(value)
    if isinstance(value, Path):
        return str(value)
    if isinstance(value, dict):
        return {str(k): to_jsonable(v) for k, v in value.items()}
    if isinstance(value, list):
        return [to_jsonable(v) for v in value]
    if isinstance(value, tuple):
        return [to_jsonable(v) for v in value]
    return value


def status(pass_value):
    if pass_value:
        return 'PASS'
    return 'FAIL'


def make_test_record(test_id, title, what_is_tested, how_it_is_tested, why_it_is_tested, expected_result, observed_result, passed, data):
    return {
        'test_id': test_id,
        'title': title,
        'vad_som_testas': what_is_tested,
        'hur_det_testas': how_it_is_tested,
        'varfor_det_testas': why_it_is_tested,
        'forvantat_resultat': expected_result,
        'observerat_resultat': observed_result,
        'slutresultat': status(passed),
        'data': to_jsonable(data)
    }


def test_1_relational_closure():
    left = Fraction(1, 3)
    right = Fraction(2, 3)
    total = left + right
    passed = total == Fraction(1, 1)
    return make_test_record(
        'TEST_1',
        'RELATIONELL_SLUTNING',
        'Forsta relationens proportion: 1/3 <-> 2/3.',
        'Beraknar 1/3 + 2/3 med exakt rationell aritmetik via fractions.Fraction.',
        'For att kontrollera att ursprungsrelationen ar sluten innan representation skapas.',
        'Summan ska vara exakt 1.',
        'Summan blev ' + str(total) + '.',
        passed,
        {
            'left': left,
            'right': right,
            'sum': total,
            'closure_target': Fraction(1, 1),
            'relation_layer': 'RELATION',
            'may_claim_origin': True
        }
    )


def test_2_representational_closure():
    reference = Fraction(1, 4)
    active = Fraction(3, 4)
    total = reference + active
    passed = total == Fraction(1, 1)
    return make_test_record(
        'TEST_2',
        'REPRESENTATIONELL_SLUTNING',
        'Forsta representationens proportion: 1/4 <-> 3/4.',
        'Beraknar 1/4 + 3/4 med exakt rationell aritmetik via fractions.Fraction.',
        'For att kontrollera att representationen fortfarande beskriver samma slutna helhet.',
        'Summan ska vara exakt 1.',
        'Summan blev ' + str(total) + '.',
        passed,
        {
            'reference': reference,
            'active': active,
            'sum': total,
            'closure_target': Fraction(1, 1),
            'representation_layer': 'REPRESENTATION',
            'may_claim_origin': False
        }
    )


def test_3_rx8_capsulation():
    r_symbol = 'R'
    diameter = Fraction(2, 1)
    square_sides = 4
    perimeter_coefficient = diameter * square_sides
    expected = Fraction(8, 1)
    passed = perimeter_coefficient == expected
    return make_test_record(
        'TEST_3',
        'GEOMETRISK_KAPSLING_RX8',
        'Rx8 som kapsling och rekonstruktion: 2R x 4 = 8R.',
        'Beraknar kvadratens fyra sidor som 4 stycken diameterlinjer, dar varje diameter ar 2R.',
        'For att kontrollera att Rx8 uppstar som strukturell kapsling av R och inte som ny ursprungsrelation.',
        'Perimeterkoefficienten ska vara exakt 8.',
        'Perimeterkoefficienten blev ' + str(perimeter_coefficient) + ', dvs 8R.',
        passed,
        {
            'R_status': 'symbolic_not_measured',
            'R_symbol': r_symbol,
            'diameter': '2R',
            'square_sides': square_sides,
            'calculation': '2R * 4 = 8R',
            'perimeter_coefficient': perimeter_coefficient,
            'expected_coefficient': expected,
            'rx8_layer': 'STRUCTURE_RECONSTRUCTION',
            'may_claim_origin': False
        }
    )


def test_4_proportion_preservation():
    exact_active = Fraction(3, 4)
    exact_reference = Fraction(1, 4)
    decimal_active = 0.75
    decimal_reference = 0.25
    active_preserved = float(exact_active) == decimal_active
    reference_preserved = float(exact_reference) == decimal_reference
    sum_preserved = math.isclose(decimal_active + decimal_reference, 1.0, rel_tol=0.0, abs_tol=0.0)
    passed = active_preserved and reference_preserved and sum_preserved
    return make_test_record(
        'TEST_4',
        'PROPORTIONSBEVARANDE',
        'Bevarande av 3/4 <-> 1/4 genom decimal representation: 0.75 <-> 0.25.',
        'Jamfor exakt brakaritmetik mot decimal/flyttalsrepresentation for 3/4 och 1/4.',
        'For att kontrollera att representationen visar proportionen utan att andra relationer skapas.',
        '3/4 ska representeras som 0.75, 1/4 som 0.25, och summan ska vara 1.0.',
        'Observerat: 0.75 + 0.25 = ' + str(decimal_active + decimal_reference) + '.',
        passed,
        {
            'exact_active': exact_active,
            'exact_reference': exact_reference,
            'decimal_active': decimal_active,
            'decimal_reference': decimal_reference,
            'active_preserved': active_preserved,
            'reference_preserved': reference_preserved,
            'sum_preserved': sum_preserved,
            'representation_creates_new_relation': False
        }
    )


def test_5_residual_position():
    exact_ratio = Fraction(3, 4)
    exact_lock = Fraction(1, 4)
    ri_float = math.sqrt(3.0) / 2.0
    observed_ratio = ri_float * ri_float
    observed_lock = 1.0 - observed_ratio
    expected_ratio = float(exact_ratio)
    expected_lock = float(exact_lock)
    ratio_residual = observed_ratio - expected_ratio
    lock_residual = observed_lock - expected_lock
    residual_sum = ratio_residual + lock_residual
    symmetric_residual = math.isclose(residual_sum, 0.0, rel_tol=0.0, abs_tol=1e-30)
    exact_relation_intact = exact_ratio + exact_lock == Fraction(1, 1)
    residual_claims_origin = False
    passed = symmetric_residual and exact_relation_intact and not residual_claims_origin
    return make_test_record(
        'TEST_5',
        'RESIDUALPOSITION',
        'Residualens lagerposition: ideal proportion minus maskinell representation.',
        'Beraknar sqrt(3)/2 i float, kvadrerar den, jamfor mot exakt 3/4 och isolerar residualparet.',
        'For att kontrollera att residualen ar ett representationsspar och inte en ny relation.',
        'Residualerna ska vara symmetriska och den exakta relationen 3/4 + 1/4 ska forbli intakt.',
        'Ratio residual ' + format(ratio_residual, '.20e') + ', lock residual ' + format(lock_residual, '.20e') + '.',
        passed,
        {
            'exact_ratio': exact_ratio,
            'exact_lock': exact_lock,
            'exact_sum': exact_ratio + exact_lock,
            'ri_float': format(ri_float, '.20e'),
            'observed_ratio_float': format(observed_ratio, '.20e'),
            'observed_lock_float': format(observed_lock, '.20e'),
            'expected_ratio_float': format(expected_ratio, '.20e'),
            'expected_lock_float': format(expected_lock, '.20e'),
            'ratio_residual': format(ratio_residual, '.20e'),
            'lock_residual': format(lock_residual, '.20e'),
            'residual_sum': format(residual_sum, '.20e'),
            'symmetric_residual': symmetric_residual,
            'exact_relation_intact': exact_relation_intact,
            'residual_layer': 'REPRESENTATION_TRACE',
            'residual_claims_origin': residual_claims_origin
        }
    )


def test_6_global_proportional_invariance(previous_tests):
    failed = [t for t in previous_tests if t['slutresultat'] != 'PASS']
    passed = len(failed) == 0
    chain = [
        'RELATION: 1/3 <-> 2/3',
        'REPRESENTATION: 1/4 <-> 3/4',
        'STRUCTURE: 2R x 4 = 8R',
        'PROPORTION: 3/4 <-> 1/4 preserved as 0.75 <-> 0.25',
        'RESIDUAL: representation trace, not origin'
    ]
    return make_test_record(
        'TEST_6',
        'GLOBAL_PROPORTIONELL_INVARIANS',
        'Hela proportionskedjan fran relation till residual.',
        'Sammanstaller status fran TEST_1 till TEST_5 och kraver att samtliga ar PASS.',
        'For att kontrollera att ingen del av kedjan bryter proportionen eller gor sig sjalv till ursprung.',
        'Alla tidigare tester ska vara PASS.',
        'Antal misslyckade tidigare tester: ' + str(len(failed)) + '.',
        passed,
        {
            'chain': chain,
            'failed_tests': [t['test_id'] for t in failed],
            'all_previous_tests_passed': passed,
            'no_later_layer_claims_origin': passed
        }
    )

def build_control_log(summary, tests, out_dir):
    lines = []
    lines.append('RP9 RX8 PROPORTION STRUCTURE VALIDATOR CONTROL LOG')
    lines.append('Generated: ' + summary['generated_at'])
    lines.append('Script: ' + summary['script'])
    lines.append('Version: ' + summary['version'])
    lines.append('Output directory: ' + str(out_dir))
    lines.append('Global status: ' + summary['global_status'])
    lines.append('')
    for test in tests:
        lines.append(test['test_id'] + ' | ' + test['slutresultat'] + ' | ' + test['title'])
        lines.append('WHAT: ' + test['vad_som_testas'])
        lines.append('HOW: ' + test['hur_det_testas'])
        lines.append('WHY: ' + test['varfor_det_testas'])
        lines.append('RESULT: ' + test['observerat_resultat'])
        lines.append('')
    return '\n'.join(lines)


def run_validator():
    out_dir = make_output_dir()
    tests = []
    tests.append(test_1_relational_closure())
    tests.append(test_2_representational_closure())
    tests.append(test_3_rx8_capsulation())
    tests.append(test_4_proportion_preservation())
    tests.append(test_5_residual_position())
    tests.append(test_6_global_proportional_invariance(tests))
    global_pass = all(test['slutresultat'] == 'PASS' for test in tests)
    global_status = status(global_pass)
    summary = {
        'system': 'RP9_RX8',
        'phase': 'FAS_2_PROPORTION_STRUCTURE_VALIDATION',
        'script': SCRIPT_NAME,
        'version': SCRIPT_VERSION,
        'generated_at': now_iso(),
        'global_status': global_status,
        'principle': 'Relation and proportion are primary through form and structure.',
        'validation_boundary': 'No external reality layer. Only relation, proportion, form, structure and residual position are tested.'
    }
    payload = {
        'SUMMARY': summary,
        'TESTS': tests
    }
    trace_payload = {
        'TRACE': tests
    }
    audit_payload = {
        'AUDIT': {
            'all_tests_include_what_how_why_result': all(
                all(key in test for key in ['vad_som_testas', 'hur_det_testas', 'varfor_det_testas', 'slutresultat'])
                for test in tests
            ),
            'no_external_reality_layer': True,
            'proportion_structure_validation_only': True,
            'later_layer_origin_claim_detected': False if global_pass else 'SEE_FAILED_TESTS',
            'global_status': global_status
        }
    }
    verify_payload = {
        'VERIFY': {
            'test_1_relational_closure': tests[0]['slutresultat'] == 'PASS',
            'test_2_representational_closure': tests[1]['slutresultat'] == 'PASS',
            'test_3_rx8_capsulation': tests[2]['slutresultat'] == 'PASS',
            'test_4_proportion_preservation': tests[3]['slutresultat'] == 'PASS',
            'test_5_residual_position': tests[4]['slutresultat'] == 'PASS',
            'test_6_global_invariance': tests[5]['slutresultat'] == 'PASS',
            'global_status': global_status
        }
    }
    write_json(out_dir / 'rp9_rx8_proportion_structure_data.json', payload)
    write_json(out_dir / 'rp9_rx8_proportion_structure_trace.json', trace_payload)
    write_json(out_dir / 'rp9_rx8_proportion_structure_audit.json', audit_payload)
    write_json(out_dir / 'rp9_rx8_proportion_structure_verify.json', verify_payload)
    markdown = build_markdown(summary, tests)
    write_text(out_dir / 'rp9_rx8_proportion_structure_presentation.md', markdown)
    control_log = build_control_log(summary, tests, out_dir)
    write_text(out_dir / 'rp9_rx8_proportion_structure_control_log.txt', control_log)
    print('=' * 72)
    print('RP9 RX8 PROPORTION STRUCTURE VALIDATOR')
    print('STATUS:', global_status)
    print('OUTPUT:', out_dir)
    print('=' * 72)
    for test in tests:
        print(test['test_id'] + ' | ' + test['slutresultat'] + ' | ' + test['title'])
    print('=' * 72)
    if not global_pass:
        return 1
    return 0


if __name__ == '__main__':
    exit_code = run_validator()
    sys.exit(exit_code)
