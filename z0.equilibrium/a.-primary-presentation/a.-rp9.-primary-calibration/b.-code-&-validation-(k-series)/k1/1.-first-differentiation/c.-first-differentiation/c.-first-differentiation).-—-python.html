def decimal_from_fraction(value, places):
    d = Decimal(value.numerator) / Decimal(value.denominator)
    return format(d, "." + str(places) + "f")


def sci(value, digits=20):
    return format(float(value), "." + str(digits) + "e")


def json_safe(obj):
    if isinstance(obj, dict):
        return {str(k): json_safe(v) for k, v in obj.items()}
    if isinstance(obj, list):
        return [json_safe(v) for v in obj]
    if isinstance(obj, tuple):
        return [json_safe(v) for v in obj]
    if isinstance(obj, Path):
        return str(obj)
    if isinstance(obj, Fraction):
        return str(obj)
    if isinstance(obj, Decimal):
        return str(obj)
    if isinstance(obj, bool):
        return bool(obj)
    if isinstance(obj, int):
        return int(obj)
    if isinstance(obj, float):
        return float(obj)
    if obj is None:
        return None
    return obj


def write_text(path, lines):
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")


def write_json(path, payload):
    path.write_text(json.dumps(json_safe(payload), indent=2, ensure_ascii=False), encoding="utf-8")


def sha256_file(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        while True:
            chunk = f.read(1024 * 1024)
            if not chunk:
                break
            h.update(chunk)
    return h.hexdigest()

def compute_exact_ratio_test(trace):

    r = Fraction(1, 1)
    d = Fraction(1, 1)
    ri_squared = r * r - (d / 2) * (d / 2)
    ratio_exact = ri_squared / (r * r)
    lock_exact = Fraction(1, 1) - ratio_exact
    ratio_float = float(ratio_exact)
    lock_float = float(lock_exact)
    ratio_residual = Fraction.from_float(ratio_float) - ratio_exact
    lock_residual = Fraction.from_float(lock_float) - lock_exact
    passed = ratio_exact == Fraction(3, 4) and lock_exact == Fraction(1, 4) and ratio_residual == 0 and lock_residual == 0
    trace.append(make_trace(len(trace) + 1, layer, "exact_fraction_geometry", "R=1 ; D=1 ; ri^2=R^2-(D/2)^2", {"R": str(r), "D": str(d)}, {"ri_squared": str(ri_squared)}, "PASS", "Uses Fraction arithmetic only."))

def compute_ieee_probe(trace):
    layer = "C_IEEE_RESIDUAL_PROBE"
    values = []
    cases = [
        ("decimal_addition", "0.1 + 0.2", 0.1 + 0.2, 0.3, "Classic binary float decimal representation check."),
        ("sin_pi", "math.sin(math.pi)", math.sin(math.pi), 0.0, "Checks transcendental rendering near zero."),
        ("tan_pi", "math.tan(math.pi)", math.tan(math.pi), 0.0, "Checks tangent rendering near zero at pi."),
        ("sqrt2_squared_minus_2", "math.sqrt(2)**2 - 2", math.sqrt(2) ** 2 - 2.0, 0.0, "Checks square-root roundtrip residual.")
    ]
    for name, expression, observed, expected, note in cases:
        residual = observed - expected
        is_close = math.isclose(observed, expected, rel_tol=0.0, abs_tol=1e-15)
        row = {"name": name, "expression": expression, "observed": sci(observed), "expected": sci(expected), "residual": sci(residual), "is_close_abs_1e_minus_15": bool(is_close), "note": note}
        values.append(row)
        trace.append(make_trace(len(trace) + 1, layer, name, expression, {}, row, "PASS" if is_close else "REVIEW", note))
    eps = sys.float_info.epsilon
    r16 = eps
    r15 = eps * 8.0
    quotient = r15 / r16
    quotient_passed = bool(quotient == 8.0)
    next_value = math.nextafter(1.0, 2.0)
    next_diff = next_value - 1.0
    next_passed = bool(next_diff == r16)
    trace.append(make_trace(len(trace) + 1, layer, "machine_epsilon", "sys.float_info.epsilon", {}, {"epsilon": sci(eps), "power_relation": "2^-52"}, "PASS", "Reads Python float machine epsilon."))
    trace.append(make_trace(len(trace) + 1, layer, "r15_r16_ratio", "(epsilon * 8) / epsilon", {"r16": sci(r16), "r15": sci(r15)}, {"quotient": str(quotient)}, "PASS" if quotient_passed else "FAIL", "Verifies R15/R16 quotient in binary precision."))
    trace.append(make_trace(len(trace) + 1, layer, "nextafter", "math.nextafter(1.0, 2.0) - 1.0", {"start": "1.0", "direction": "2.0"}, {"next_value": sci(next_value), "diff": sci(next_diff)}, "PASS" if next_passed else "FAIL", "Checks the next representable float above 1.0."))
    passed = quotient_passed and next_passed
    return {
        "layer_name": layer,
        "purpose": "Control group for IEEE-754 style float behaviour using Python float.",
        "float_cases": values,
        "machine_precision": {
            "epsilon": sci(eps),
            "R16": sci(r16),
            "R15": sci(r15),
            "R15_div_R16": str(quotient),
            "R15_div_R16_passed": quotient_passed,
            "nextafter_1_to_2": sci(next_value),
            "nextafter_diff": sci(next_diff),
            "nextafter_matches_R16": next_passed
        },
        "status": "PASS" if passed else "REVIEW"
    }


def compute_final_float_lock(trace):
    layer = "D_FINAL_FLOAT_LOCK"
    r = 1.0
    d = 1.0
    ri = math.sqrt((r * r) - ((d / 2.0) * (d / 2.0)))
    ay = math.pi * r * r
    ai = math.pi * ri * ri
    ratio = ai / ay
    lock = 1.0 - ratio
    ratio_target = 0.75
    lock_target = 0.25
    ratio_residual = ratio - ratio_target
    lock_residual = lock - lock_target
    ratio_next_down = math.nextafter(ratio, 0.0)
    ratio_next_up = math.nextafter(ratio, 1.0)
    lock_next_down = math.nextafter(lock, 0.0)
    lock_next_up = math.nextafter(lock, 1.0)
    symmetric_residual = bool(abs(ratio_residual + lock_residual) <= 1e-30)
    ratio_is_lower = bool(ratio < ratio_target)
    lock_is_higher = bool(lock > lock_target)
    confirmed = symmetric_residual and ratio_is_lower and lock_is_higher
    trace.append(make_trace(len(trace) + 1, layer, "float_geometry", "ri = sqrt(R^2 - (D/2)^2)", {"R": sci(r), "D": sci(d)}, {"RI": sci(ri)}, "PASS", "Runs the same geometry through binary float path."))
    trace.append(make_trace(len(trace) + 1, layer, "float_area_ratio", "ratio = (pi * RI^2) / (pi * R^2)", {"AI": sci(ai), "AY": sci(ay)}, {"ratio": sci(ratio), "lock": sci(lock)}, "PASS", "Computes visible float rendering of 3/4 and 1/4."))
    trace.append(make_trace(len(trace) + 1, layer, "float_lock_residual", "ratio - 0.75 ; lock - 0.25", {"ratio": sci(ratio), "lock": sci(lock)}, {"ratio_residual": sci(ratio_residual), "lock_residual": sci(lock_residual)}, "PASS" if confirmed else "REVIEW", "Checks symmetric float lock residual."))
    classification = "SYMMETRIC_FLOAT_LOCK_CONFIRMED" if confirmed else "FLOAT_LOCK_REVIEW_REQUIRED"
    return {
        "layer_name": layer,
        "purpose": "Final float-path confirmation of the locked differential rendering.",
        "float_geometry": {"R": sci(r), "D": sci(d), "RI": sci(ri), "AY": sci(ay), "AI": sci(ai)},
        "float_lock": {
            "ratio_observed": sci(ratio),
            "ratio_expected": sci(ratio_target),
            "ratio_residual": sci(ratio_residual),
            "ratio_hex": ratio.hex(),
            "ratio_next_down": sci(ratio_next_down),
            "ratio_next_up": sci(ratio_next_up),
            "lock_observed": sci(lock),
            "lock_expected": sci(lock_target),
            "lock_residual": sci(lock_residual),
            "lock_hex": lock.hex(),
            "lock_next_down": sci(lock_next_down),
            "lock_next_up": sci(lock_next_up)
        },
        "classification": {"ratio_is_lower_than_0_75": ratio_is_lower, "lock_is_higher_than_0_25": lock_is_higher, "symmetric_residual": symmetric_residual, "classification": classification},
        "status": "PASS" if confirmed else "REVIEW"
    }


def build_master_conclusion(layer_a, layer_b, layer_c, layer_d):
    statuses = [layer_a["status"], layer_b["status"], layer_c["status"], layer_d["status"]]
    if any(status == "FAIL" for status in statuses):

        total = "PASS"

        "layer_name": "E_MASTER_CONCLUSION",

        "summary": {"first_structure": layer_a["status"], "exact_ratio_test": layer_b["status"], "ieee_probe": layer_c["status"], "final_float_lock": layer_d["status"], "total_status": total},
        "measurement_boundary": "The measurement verifies exact geometry and machine representation behaviour. RP9 interpretation is separated below.",

    }

    lines.append("CONTROL RESULT")
    lines.append("-" * 72)
    for key, value in e["summary"].items():
        lines.append(key + ": " + str(value))
    lines.append("")
    lines.append("BOUNDARY")
    lines.append("-" * 72)
    lines.append(e["measurement_boundary"])
    return lines


def build_execution_log(payload, paths):
    lines = []
    lines.append("FIRST DIFFERENTIATION EXECUTION LOG")
    lines.append("=" * 72)
    lines.append("Timestamp local: " + now_local())
    lines.append("Timestamp UTC: " + now_utc_iso())
    lines.append("Status: " + payload["layers"]["E_master_conclusion"]["summary"]["total_status"])

def write_outputs(payload, trace, audit_entries):
    ensure_output_dir()
    data_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_DATA.json"
    trace_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_TRACE.json"
    full_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_FULL_DATASHEET.json"
    md_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_PRESENTATION.md"
    control_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_CONTROL_LOG.txt"
    execution_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_EXECUTION.log"
    audit_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_AUDIT.json"
    verify_path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_VERIFY.json"
    append_audit(audit_entries, "write_data_json", "ok", {"path": str(data_path)})
    write_json(data_path, payload)
    append_audit(audit_entries, "write_trace_json", "ok", {"path": str(trace_path)})
    write_json(trace_path, trace)
    append_audit(audit_entries, "write_full_datasheet_json", "ok", {"path": str(full_path)})
    write_json(full_path, {"payload": payload, "trace": trace})
    append_audit(audit_entries, "write_markdown", "ok", {"path": str(md_path)})
    write_text(md_path, build_markdown(payload, trace))
    append_audit(audit_entries, "write_control_log", "ok", {"path": str(control_path)})
    write_text(control_path, build_control_log(payload, trace))
    paths = {"data_json": data_path, "trace_json": trace_path, "full_datasheet_json": full_path, "presentation_markdown": md_path, "control_log": control_path, "execution_log": execution_path, "audit_json": audit_path, "verify_json": verify_path}
    append_audit(audit_entries, "write_execution_log", "ok", {"path": str(execution_path)})
    write_text(execution_path, build_execution_log(payload, paths))
    audit_payload = {"script_name": SCRIPT_NAME, "script_version": SCRIPT_VERSION, "created_at_utc": now_utc_iso(), "entry_count": len(audit_entries), "entries": audit_entries}
    write_json(audit_path, audit_payload)
    verify_files = {}
    for name, path in paths.items():
        if path.exists() and path != verify_path:
            verify_files[name] = {"path": str(path), "bytes": path.stat().st_size, "sha256": sha256_file(path)}
    verify_payload = {"script_name": SCRIPT_NAME, "script_version": SCRIPT_VERSION, "generated_at_utc": now_utc_iso(), "purpose": "Verification hashes for generated FIRST DIFFERENTIATION artifacts.", "files": verify_files}
    write_json(verify_path, verify_payload)
    return paths

def generate_once():
    audit_entries = []
    append_audit(audit_entries, "start_generation", "ok", {"script_name": SCRIPT_NAME, "script_version": SCRIPT_VERSION, "working_directory": str(BASE_DIR), "python_version": sys.version, "platform": platform.platform()})
    payload, trace = run_all()

def write_error_log(exc):
    ensure_output_dir()
    path = OUTPUT_DIR / "FIRST_DIFFERENTIATION_ERROR.log"
    lines = []
    lines.append("FIRST DIFFERENTIATION ERROR LOG")
    lines.append("=" * 72)
    lines.append("Timestamp local: " + now_local())
    lines.append("Timestamp UTC: " + now_utc_iso())
    lines.append("Error type: " + type(exc).__name__)
    lines.append("Error message: " + str(exc))
    lines.append("")
    lines.append("TRACEBACK")
    lines.append("-" * 72)
    lines.extend(traceback.format_exc().splitlines())
    write_text(path, lines)
    return path


def parse_args():
    parser = argparse.ArgumentParser(description="RP9 FIRST DIFFERENTIATION all-inclusive presentation generator.")
    parser.add_argument("--run-once", action="store_true", help="Generate all files once and exit without menu.")
    return parser.parse_args()


def main():
    args = parse_args()
    try:
        if args.run_once:
            generate_once()
        else:
            interactive_menu()
    except KeyboardInterrupt:
        print("")
        print("Avbrutet av användaren.")
    except Exception as exc:
        error_path = write_error_log(exc)
        print("ERROR:", str(exc))
        print("Error log written to:", error_path)
    finally:
        if not args.run_once:
            input("Tryck Enter för att stänga...")


if __name__ == "__main__":
    main()
