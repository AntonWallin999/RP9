# RP9_RESIDUAL_ALL_IN_ONE_PRESENTATION.py
# Version: 2.0
# File type: Python
# Purpose:
# - All-in-one residual verification for R15 / R16.
# - Writes JSON data, trace JSON, Markdown report, control log and execution log.
# - Uses standard library only. No numpy dependency.
# - Keeps measurement and RP9 interpretation separated for review.

import json
import datetime
import math
import sys
from pathlib import Path

SCRIPT_VERSION = 'RP9_RESIDUAL_VALUE'
OUTPUT_DIR_NAME = 'RP9_RESIDUAL_VALUE'


def get_base_dir():
    try:
        return Path(__file__).resolve().parent
    except NameError:
        return Path.cwd()


BASE_DIR = get_base_dir()
OUTPUT_DIR = BASE_DIR / OUTPUT_DIR_NAME


def now_stamp():
    return datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')


def iso_stamp():
    return datetime.datetime.now().isoformat(timespec='seconds')


def ensure_output_dir():
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)


def as_float_text(value):
    return format(float(value), '.20e')


def as_plain_text(value):
    return str(value)


def json_safe(value):
    if isinstance(value, dict):
        return {str(k): json_safe(v) for k, v in value.items()}
    if isinstance(value, list):
        return [json_safe(v) for v in value]
    if isinstance(value, tuple):
        return [json_safe(v) for v in value]
    if isinstance(value, Path):
        return str(value)
    if isinstance(value, bool):
        return bool(value)
    if isinstance(value, int):
        return int(value)
    if isinstance(value, float):
        if math.isnan(value):
            return 'NaN'
        if math.isinf(value):
            return 'Infinity' if value > 0 else '-Infinity'
        return float(value)
    return value


def write_text_file(path, lines):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text('\n'.join(lines) + '\n', encoding='utf-8')


def write_json_file(path, payload):
    path.parent.mkdir(parents=True, exist_ok=True)
    safe_payload = json_safe(payload)
    path.write_text(json.dumps(safe_payload, indent=2, ensure_ascii=False), encoding='utf-8')


def make_trace_step(step, operation, formula, inputs, result, status, note):
    return {
        'step': int(step),
        'timestamp': iso_stamp(),
        'operation': operation,
        'formula': formula,
        'inputs': inputs,
        'result': result,
        'status': status,
        'note': note
    }


def run_measurements():
    trace = []

    eps = sys.float_info.epsilon
    trace.append(make_trace_step(
        1,
        'read_machine_epsilon',
        'sys.float_info.epsilon',
        {},
        {
            'epsilon_decimal': as_plain_text(eps),
            'epsilon_scientific': as_float_text(eps),
            'epsilon_expected_power': '2^-52'
        },
        'PASS',
        'Reads IEEE754 double precision machine epsilon from Python standard library.'
    ))

    r16 = eps
    trace.append(make_trace_step(
        2,
        'assign_r16',
        'R16 = epsilon',
        {'epsilon': as_float_text(eps)},
        {
            'r16': as_float_text(r16),
            'r16_power': '2^-52'
        },
        'PASS',
        'Defines R16 as machine epsilon for double precision.'
    ))

    r15 = r16 * 8.0
    trace.append(make_trace_step(
        3,
        'calculate_r15',
        'R15 = R16 * 8',
        {
            'r16': as_float_text(r16),
            'factor': '8'
        },
        {
            'r15': as_float_text(r15),
            'r15_power': '2^-49'
        },
        'PASS',
        'Calculates R15 as three binary steps larger than R16.'
    ))

    quotient = r15 / r16
    quotient_passed = bool(quotient == 8.0)
    trace.append(make_trace_step(
        4,
        'calculate_quotient',
        'R15 / R16',
        {
            'r15': as_float_text(r15),
            'r16': as_float_text(r16)
        },
        {
            'quotient': as_plain_text(quotient),
            'expected': '8.0'
        },
        'PASS' if quotient_passed else 'FAIL',
        'Verifies that the R15/R16 relation gives exact quotient 8.0.'
    ))

    power_52 = 2.0 ** -52
    power_49 = 2.0 ** -49
    power_quotient = power_49 / power_52
    power_passed = bool(power_quotient == 8.0)
    trace.append(make_trace_step(
        5,
        'direct_power_verification',
        '(2.0**-49) / (2.0**-52)',
        {
            '2^-52': as_float_text(power_52),
            '2^-49': as_float_text(power_49)
        },
        {
            'quotient': as_plain_text(power_quotient),
            'expected': '8.0'
        },
        'PASS' if power_passed else 'FAIL',
        'Direct verification using powers of two.'
    ))

    x = 1.0
    next_x = math.nextafter(x, 2.0)
    diff_r16 = next_x - x
    nextafter_passed = bool(diff_r16 == r16)
    trace.append(make_trace_step(
        6,
        'nextafter_practical_verification',
        'math.nextafter(1.0, 2.0) - 1.0',
        {
            'x': '1.0',
            'direction': '2.0'
        },
        {
            'next_value': as_float_text(next_x),
            'diff': as_float_text(diff_r16),
            'expected_r16': as_float_text(r16)
        },
        'PASS' if nextafter_passed else 'FAIL',
        'Practical verification of the smallest representable step above 1.0.'
    ))

    nextafter_series = []
    x_series = 1.0
    for i in range(1, 6):
        next_value = math.nextafter(x_series, 2.0)
        diff = next_value - x_series
        nextafter_series.append({
            'step': int(i),
            'input_value': as_float_text(x_series),
            'next_value': as_float_text(next_value),
            'diff': as_float_text(diff)
        })
        x_series = next_value

    trace.append(make_trace_step(
        7,
        'nextafter_series',
        'repeat math.nextafter(x, 2.0) for 5 steps',
        {
            'start': '1.0',
            'steps': '5'
        },
        {'series': nextafter_series},
        'PASS',
        'Shows several consecutive representable float values around 1.0.'
    ))

    scale_series = []
    for exp in range(-3, 4):
        value = 1.0 * (2.0 ** exp)
        local_eps_estimate = eps * abs(value)
        scale_series.append({
            'power': '2^' + str(exp),
            'value': as_float_text(value),
            'local_epsilon_estimate': as_float_text(local_eps_estimate)
        })

    trace.append(make_trace_step(
        8,
        'scale_demo',
        'local_eps = sys.float_info.epsilon * abs(value)',
        {'range': '2^-3 to 2^3'},
        {'series': scale_series},
        'PASS',
        'Shows how machine epsilon scales with magnitude as an estimate.'
    ))

    total_passed = bool(quotient_passed and power_passed and nextafter_passed)

    payload = {
        'script': {
            'name': SCRIPT_VERSION,
            'created_at': now_stamp(),
            'purpose': 'All-in-one presentation and verification code for RP9 residual R15/R16.',
            'method': 'IEEE754 double precision inspection using Python standard library.',
            'review_design': 'Measurement, trace, presentation, control log and RP9 interpretation are separated.',
            'dependency_note': 'No external packages required.'
        },
        'measurement': {
            'name': 'R15_R16_RESIDUAL_VERIFICATION',
            'machine_epsilon': {
                'value': as_float_text(eps),
                'definition': 'sys.float_info.epsilon',
                'power_relation': '2^-52'
            },
            'r16': {
                'value': as_float_text(r16),
                'definition': 'R16 = epsilon',
                'power_relation': '2^-52'
            },
            'r15': {
                'value': as_float_text(r15),
                'definition': 'R15 = R16 * 8',
                'power_relation': '2^-49'
            },
            'quotient': {
                'formula': 'R15 / R16',
                'value': as_plain_text(quotient),
                'expected': '8.0',
                'passed': bool(quotient_passed)
            },
            'direct_power_check': {
                'formula': '(2.0**-49) / (2.0**-52)',
                'value': as_plain_text(power_quotient),
                'expected': '8.0',
                'passed': bool(power_passed)
            },
            'nextafter_check': {
                'formula': 'math.nextafter(1.0, 2.0) - 1.0',
                'value': as_float_text(diff_r16),
                'expected': as_float_text(r16),
                'passed': bool(nextafter_passed)
            },
            'nextafter_series': nextafter_series,
            'scale_series': scale_series,
            'total_status': 'PASS' if total_passed else 'FAIL'
        },
        'relation_chain': {
            'r16': '2^-52',
            'r15': '2^-49',
            'binary_step_difference': '3',
            'ratio': '2^3 = 8',
            'review_note': 'This is a machine precision relation measured inside IEEE754 double precision.'
        },
        'rp9_interpretation_layer': {
            'separated_from_measurement': True,
            'interpretation': 'The quotient 8.0 can be related to octal structure and Base 8 as an RP9 interpretation.',
            'important_boundary': 'The measurement verifies R15/R16 = 8.0. The RP9 interpretation is a separate layer.'
        }
    }

    return payload, trace

def write_all_outputs(payload, trace):
    ensure_output_dir()

    json_path = OUTPUT_DIR / 'rp9_residual_data.json'
    trace_path = OUTPUT_DIR / 'rp9_residual_trace.json'
    markdown_path = OUTPUT_DIR / 'rp9_residual_presentation.md'
    control_path = OUTPUT_DIR / 'rp9_residual_control_log.txt'
    execution_path = OUTPUT_DIR / 'rp9_residual_execution.log'

    write_json_file(json_path, payload)
    write_json_file(trace_path, trace)
    write_text_file(markdown_path, build_markdown(payload))
    write_text_file(control_path, build_control_log(payload, trace))

    log_lines = []
    log_lines.append('RP9 RESIDUAL EXECUTION LOG')
    log_lines.append('=' * 72)
    log_lines.append('Timestamp: ' + now_stamp())
    log_lines.append('Status: ' + payload['measurement']['total_status'])
    log_lines.append('JSON: ' + str(json_path))
    log_lines.append('Trace: ' + str(trace_path))
    log_lines.append('Markdown: ' + str(markdown_path))
    log_lines.append('Control log: ' + str(control_path))
    write_text_file(execution_path, log_lines)

    return {
        'json': json_path,
        'trace': trace_path,
        'markdown': markdown_path,
        'control': control_path,
        'execution': execution_path
    }


def print_summary(payload):
    m = payload['measurement']
    print('')
    print('=' * 72)
    print('RP9 RESIDUAL SUMMARY')
    print('=' * 72)
    print('R16:', m['r16']['value'])
    print('R15:', m['r15']['value'])
    print('R15 / R16:', m['quotient']['value'])
    print('Direct power check:', m['direct_power_check']['value'])
    print('Nextafter diff:', m['nextafter_check']['value'])
    print('Total status:', m['total_status'])
    print('=' * 72)
    print('')


def print_nextafter(payload):
    print('')
    print('=' * 72)
    print('NEXTAFTER SERIES AROUND 1.0')
    print('=' * 72)
    for row in payload['measurement']['nextafter_series']:
        print('Step', row['step'])
        print('  Input:', row['input_value'])
        print('  Next :', row['next_value'])
        print('  Diff :', row['diff'])
    print('=' * 72)
    print('')


def print_scale(payload):
    print('')
    print('=' * 72)
    print('SCALE DEMO')
    print('=' * 72)
    for row in payload['measurement']['scale_series']:
        print(row['power'], '-> value:', row['value'], '| local eps:', row['local_epsilon_estimate'])
    print('=' * 72)
    print('')


def print_paths(paths):
    print('')
    print('FILES WRITTEN')
    print('-' * 72)
    print('JSON data     :', paths['json'])
    print('Trace JSON    :', paths['trace'])
    print('Markdown      :', paths['markdown'])
    print('Control log   :', paths['control'])
    print('Execution log :', paths['execution'])
    print('')


def run_all():
    payload, trace = run_measurements()
    paths = write_all_outputs(payload, trace)
    return payload, trace, paths


def interactive_menu():
    payload, trace, paths = run_all()
    print_summary(payload)
    print_paths(paths)

    while True:
        print('')
        print('=' * 72)
        print('RP9 - ALL IN ONE RESIDUAL PRESENTATION')
        print('=' * 72)
        print('1. Visa grundsammanfattning')
        print('2. Visa nextafter-serie runt 1.0')
        print('3. Visa skalningsdemonstration')
        print('4. Visa alla')
        print('5. Skriv om alla exportfiler')
        print('6. Visa exportvagar')
        print('0. Avsluta')
        print('')

        choice = input('Ditt val (0-6): ').strip()

        if choice == '1':
            print_summary(payload)
        elif choice == '2':
            print_nextafter(payload)
        elif choice == '3':
            print_scale(payload)
        elif choice == '4':
            print_summary(payload)
            print_nextafter(payload)
            print_scale(payload)
            print_paths(paths)
        elif choice == '5':
            payload, trace, paths = run_all()
            print('Alla exportfiler skrevs om.')
            print_paths(paths)
        elif choice == '6':
            print_paths(paths)
        elif choice == '0':
            print('Avslutar.')
            break
        else:
            print('Ogiltigt val.')

        input('Tryck Enter for att fortsatta...')


def run_once_mode():
    payload, trace, paths = run_all()
    print_summary(payload)
    print_paths(paths)
    return payload, trace, paths


def main():
    exit_code = 0
    try:
        if '--run-once' in sys.argv:
            run_once_mode()
        else:
            interactive_menu()
    except KeyboardInterrupt:
        print('')
        print('Avbrutet av anvandaren.')
        exit_code = 130
    except Exception as exc:
        ensure_output_dir()
        error_path = OUTPUT_DIR / 'rp9_residual_error.log'
        lines = []
        lines.append('RP9 RESIDUAL ERROR LOG')
        lines.append('=' * 72)
        lines.append('Timestamp: ' + now_stamp())
        lines.append('Error type: ' + type(exc).__name__)
        lines.append('Error message: ' + str(exc))
        write_text_file(error_path, lines)
        print('ERROR:', str(exc))
        print('Error log written to:', error_path)
        exit_code = 1
    finally:
        if '--run-once' not in sys.argv:
            input('Tryck Enter for att stanga...')
    return exit_code


if __name__ == '__main__':
    raise SystemExit(main())
