def write_json(path, data):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with Path(path).open('w', encoding='utf-8') as f:
        json.dump(data, f, indent=2, ensure_ascii=False)

def write_text(path, text):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with Path(path).open('w', encoding='utf-8') as f:
        f.write(text)

def write_csv(path, rows, fieldnames):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with Path(path).open('w', encoding='utf-8', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for row in rows:
            writer.writerow(row)

def read_json(path):
    with Path(path).open('r', encoding='utf-8') as f:
        return json.load(f)

def optional_json(path):
    if path is not None and Path(path).exists():
        try:
            return read_json(path)
        except Exception:
            return None
    return None

def find_directory(name):
    direct = ROOT_DIR / name
    if direct.exists() and direct.is_dir():
        return direct
    matches = []
    for path in ROOT_DIR.rglob(name):
        if path.is_dir() and OUTPUT_NAME not in str(path):
            matches.append(path)
    if matches:
        matches.sort(key=lambda p: (len(str(p)), str(p)))
        return matches[0]
    return None

def safe_get(data, keys, default=None):
    cur = data
    for key in keys:
        if isinstance(cur, dict) and key in cur:
            cur = cur[key]
        else:
            return default
    return cur

def vec_add(a, b):
    return [a[i] + b[i] for i in range(3)]

def vec_sub(a, b):
    return [a[i] - b[i] for i in range(3)]

def vec_mul(a, scalar):
    return [a[i] * scalar for i in range(3)]

def dot(a, b):
    return sum(a[i] * b[i] for i in range(3))

def cross(a, b):
    return [
        a[1] * b[2] - a[2] * b[1],
        a[2] * b[0] - a[0] * b[2],
        a[0] * b[1] - a[1] * b[0]
    ]

def norm(v):
    return math.sqrt(dot(v, v))

def normalize(v):
    n = norm(v)
    if n <= 1.0e-15:
        return [0.0, 0.0, 0.0]
    return [v[i] / n for i in range(3)]

def close_vec(a, b, eps=1.0e-12):
    return norm(vec_sub(a, b)) <= eps

def active_axes(v):
    axes = []
    if abs(v[0]) > 1.0e-12:
        axes.append('X')
    if abs(v[1]) > 1.0e-12:
        axes.append('Y')
    if abs(v[2]) > 1.0e-12:
        axes.append('Z')
    return axes

def get_view(name):
    for group in VIEW_GROUPS.values():
        if name in group['views']:
            return list(group['views'][name])
    raise ValueError('Unknown view: ' + str(name))

def group_for_view(view_name):
    for group_name, group in VIEW_GROUPS.items():
        if view_name in group['views']:
            return group_name
    return 'UNKNOWN'

def build_position_table():
    rows = []
    for group_name, group in VIEW_GROUPS.items():
        for view_name, vector in group['views'].items():
            normalized = normalize(vector)
            for level_name, zoom in ZOOM_LEVELS.items():
                eye = vec_mul(normalized, zoom)
                rows.append({
                    'view': view_name,
                    'group': group_name,
                    'level': level_name,
                    'zoom': zoom,
                    'raw_x': vector[0],
                    'raw_y': vector[1],
                    'raw_z': vector[2],
                    'norm_x': normalized[0],
                    'norm_y': normalized[1],
                    'norm_z': normalized[2],
                    'eye_x': eye[0],
                    'eye_y': eye[1],
                    'eye_z': eye[2],
                    'active_axes': active_axes(vector),
                    'coordinate_status': 'emergent_projected_observer_position_from_relation_vector'
                })
    return rows

def analyze_candidate_pair(pair_name, left, right):
    v1 = get_view(left)
    v2 = get_view(right)
    midpoint = vec_mul(vec_add(v1, v2), 0.5)
    is_opposite = close_vec(v1, vec_mul(v2, -1.0))
    axes = active_axes(v1)
    same_plane_xz = abs(v1[1]) <= 1.0e-12 and abs(v2[1]) <= 1.0e-12
    center_invariant = close_vec(midpoint, [0.0, 0.0, 0.0])
    delta_nonzero = norm(v1) > 1.0e-12
    three_line_lock = axes == ['X', 'Z']
    z_pass = bool(is_opposite and same_plane_xz and center_invariant and delta_nonzero and three_line_lock)
    elimination_reason = None
    if not z_pass:
        if axes == ['Y']:
            elimination_reason = 'single_Y_axis_projection_not_structural_lock'
        elif len(axes) == 1:
            elimination_reason = 'one_axis_relation_too_low_dimensional'
        elif axes == ['X', 'Y']:
            elimination_reason = 'XY_plane_rejected_Y_active'
        elif len(axes) == 3:
            elimination_reason = 'three_axis_relation_too_high_dimensional_for_2D_compression'
        else:
            elimination_reason = 'fails_three_line_lock'
    return {
        'pair': pair_name,
        'left': left,
        'right': right,
        'group_left': group_for_view(left),
        'group_right': group_for_view(right),
        'v1': v1,
        'v2': v2,
        'v1_normalized': normalize(v1),
        'v2_normalized': normalize(v2),
        'midpoint': midpoint,
        'is_opposite': is_opposite,
        'same_plane_xz': same_plane_xz,
        'center_invariant': center_invariant,
        'delta_nonzero': delta_nonzero,
        'active_axes': axes,
        'three_line_lock': three_line_lock,
        'Z_PASS': z_pass,
        'elimination_reason': elimination_reason,
        'coordinate_reconstruction_status': 'locked_from_inverse_pair_not_injected_point_geometry'
    }

def build_candidate_pair_lock():
    rows = []
    for pair_name, left, right in CANDIDATE_PAIRS:
        rows.append(analyze_candidate_pair(pair_name, left, right))
    return rows

def build_basis_from_positions(pair_rows):
    x_axis = normalize(get_view('A1'))
    y_axis = normalize(get_view('A3'))
    z_axis = normalize(get_view('A5'))
    b5b6 = None
    for row in pair_rows:
        if row['pair'] == 'B5-B6':
            b5b6 = row
            break
    basis = {
        'basis_status': 'derived_from_A_axis_inverse_pairs_then_selected_by_B5_B6_projection_lock',
        'origin': [0.0, 0.0, 0.0],
        'X_axis': x_axis,
        'Y_axis': y_axis,
        'Z_axis': z_axis,
        'orthogonality': {
            'dot_X_Y': dot(x_axis, y_axis),
            'dot_X_Z': dot(x_axis, z_axis),
            'dot_Y_Z': dot(y_axis, z_axis)
        },
        'orientation': {
            'X_cross_Y': cross(x_axis, y_axis),
            'matches_Z': close_vec(cross(x_axis, y_axis), z_axis)
        },
        'selected_projection_pair': b5b6
    }
    if b5b6:
        w = normalize(b5b6['v1'])
        ref = [0.0, 1.0, 0.0]
        u = normalize(cross(ref, w))
        v = normalize(cross(w, u))
        basis['B5_projection_basis'] = {
            'w': w,
            'u': u,
            'v': v,
            'basis_status': 'projection_basis_reconstructed_after_Z_PASS_selection'
        }
    return basis

            axis_counts['two_axis'] += 1
        elif n == 3:
            axis_counts['three_axis'] += 1

        'generated': iso_stamp(),

        'coordinate_origin_status': 'center_invariant_from_inverse_pairs',
