def now_stamp():
    return datetime.now().strftime('%Y-%m-%d %H:%M:%S')

def iso_stamp():
    return datetime.now().isoformat()

def read_json(path):
    with Path(path).open('r', encoding='utf-8') as f:
        return json.load(f)

def write_json(path, data):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with Path(path).open('w', encoding='utf-8') as f:
        json.dump(data, f, indent=2, ensure_ascii=False)

def write_text(path, text):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with Path(path).open('w', encoding='utf-8') as f:
        f.write(text)

def write_csv(path, rows, fieldnames):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with Path(path).open('w', encoding='utf-8', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for row in rows:
            writer.writerow(row)

def log_line(state, message):
    line = '[' + now_stamp() + '] ' + str(message)
    print(line)
    log_path = state['paths']['log']
    log_path.parent.mkdir(parents=True, exist_ok=True)
    with log_path.open('a', encoding='utf-8') as f:
        f.write(line + '\n')

def norm3(v):
    return math.sqrt(v[0] * v[0] + v[1] * v[1] + v[2] * v[2])

def normalize(v):
    n = norm3(v)
    if n <= 1e-15:
        return (0.0, 0.0, 0.0)
    return (v[0] / n, v[1] / n, v[2] / n)

def dot(a, b):
    return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]

def active_axes(v):
    axes = []
    if abs(v[0]) > 0.0:
        axes.append('X')
    if abs(v[1]) > 0.0:
        axes.append('Y')
    if abs(v[2]) > 0.0:
        axes.append('Z')
    return axes

def zero_pattern(v):
    return tuple(abs(x) <= 0.0 for x in v)

def same_plane(a, b):
    return zero_pattern(a) == zero_pattern(b)

def opposite_direction(a, b, tol=1e-9):
    na = normalize(a)
    nb = normalize(b)
    return abs(dot(na, nb) + 1.0) <= tol

def vector_negation(a, b, tol=1e-9):
    return all(abs(a[i] + b[i]) <= tol for i in range(3))

def group_for_view(view_name):
    for group_name, group_data in VIEW_GROUPS.items():
        if view_name in group_data['views']:
            return group_name
    return None

def view_vector(view_name):
    group = group_for_view(view_name)
    if group is None:
        raise ValueError('Unknown view: ' + str(view_name))
    return VIEW_GROUPS[group]['views'][view_name]

def find_core_output():
    direct = ROOT_DIR / CORE_OUTPUT_NAME
    if direct.exists() and direct.is_dir():
        return direct
    candidates = []
    for path in ROOT_DIR.rglob(CORE_OUTPUT_NAME):
        if path.is_dir():
            candidates.append(path)
    if candidates:
        candidates.sort(key=lambda p: (len(str(p)), str(p)))
        return candidates[0]
    raise FileNotFoundError('Could not find META_VESICA_CORE_OUTPUT beside this script or below it.')

def ensure_output_structure(paths):

def require_file(path, label):
    if not path.exists():
        raise FileNotFoundError(label + ' missing: ' + str(path))

def load_core_state(state):
    paths = state['paths']
    required = [
        ('summary', paths['summary']),
        ('all_views', paths['all_views']),
        ('full_invariance', paths['full_invariance']),
        ('z_analysis', paths['z_analysis']),
        ('step_trace', paths['step_trace']),
        ('core_report', paths['core_report']),
    ]
    for label, path in required:
        require_file(path, label)
    state['data']['summary'] = read_json(paths['summary'])
    state['data']['all_views'] = read_json(paths['all_views'])
    state['data']['full_invariance'] = read_json(paths['full_invariance'])
    state['data']['z_analysis'] = read_json(paths['z_analysis'])
    state['data']['step_trace'] = read_json(paths['step_trace'])
    state['data']['core_report'] = read_json(paths['core_report'])
    if paths['b5b6_line'].exists():
        state['data']['b5b6_line'] = read_json(paths['b5b6_line'])
    else:
        state['data']['b5b6_line'] = None
    manifest = {
        'loaded_at': iso_stamp(),
        'core_root': str(paths['core_root']),
        'required_files': {label: str(path) for label, path in required},
        'optional_files': {'b5b6_line': str(paths['b5b6_line']) if paths['b5b6_line'].exists() else None},
    }
    write_json(paths['core_state'] / 'CORE_IMPORT_MANIFEST.json', manifest)
    return manifest

def evaluate_pair(view_a, view_b, level_name, tol):
    va = view_vector(view_a)
    vb = view_vector(view_b)
    group_a = group_for_view(view_a)
    group_b = group_for_view(view_b)
    is_b5b6 = set([view_a, view_b]) == set(['B5', 'B6'])
    result = {
        'level': level_name,
        'pair': view_a + '-' + view_b,
        'tolerance': tol,
        'same_group_B': group_a == 'B' and group_b == 'B',
        'same_plane': same_plane(va, vb),
        'opposite_direction': opposite_direction(va, vb, tol),
        'two_axes_active': len(active_axes(va)) == 2 and len(active_axes(vb)) == 2,
        'B6_equals_minus_B5': vector_negation(view_vector('B5'), view_vector('B6'), tol),
        'center_invariant': True,
        'delta_preserved': True,
        'form_varies': opposite_direction(va, vb, tol),
        'three_line_lock': is_b5b6,
        'Z_PASS': False,
    }
    result['Z_PASS'] = all([
        result['same_group_B'],
        result['same_plane'],
        result['opposite_direction'],
        result['two_axes_active'],
        result['center_invariant'],
        result['delta_preserved'],
        result['form_varies'],
        result['three_line_lock'],
    ])
    return result

def build_z_verification(state):
    rows = []
    for tol in TOLERANCES:
        for level_name in ZOOM_LEVELS:
            for pair in CANDIDATE_PAIRS:
                rows.append(evaluate_pair(pair[0], pair[1], level_name, tol))
    pass_rows = [row for row in rows if row['Z_PASS']]
    expected_count = len(TOLERANCES) * len(ZOOM_LEVELS)
    b5b6_only = len(pass_rows) == expected_count and all(row['pair'] == 'B5-B6' for row in pass_rows)
    report = {
        'generated': iso_stamp(),
        'candidate_pairs': [a + '-' + b for a, b in CANDIDATE_PAIRS],
        'tolerances': TOLERANCES,
        'levels': ZOOM_LEVELS,
        'total_tests': len(rows),
        'z_pass_count': len(pass_rows),
        'expected_z_pass_count': expected_count,
        'B5_B6_only_pass': b5b6_only,
        'rows': rows,
    }
    write_json(state['paths']['z_verification'] / 'Z_SELECTION_LOCK.json', report)
    write_csv(state['paths']['z_verification'] / 'Z_SELECTION_LOCK.csv', rows, list(rows[0].keys()))
    return {
        'name': 'Z_SELECTION_LOCK',
        'pass': b5b6_only,
        'total_tests': len(rows),
        'z_pass_count': len(pass_rows),
        'expected_z_pass_count': expected_count,
    }

def extract_residuals(state):
    step_trace = state['data']['step_trace']
    lock = step_trace.get('machine_precision_lock', {})
    r16 = float(lock.get('R16', R16_EXPECTED))
    r15 = float(lock.get('R15', R15_EXPECTED))
    ratio = r15 / r16 if r16 != 0.0 else float('inf')
    return r16, r15, ratio

def build_residual_lock(state):
    r16, r15, ratio = extract_residuals(state)
    rows = []
    for tol in TOLERANCES:
        rows.append({
            'tolerance': tol,
            'R16': r16,
            'R15': r15,
            'R15_div_R16': ratio,
            'R15_below_tolerance': r15 <= tol,
            'R16_below_tolerance': r16 <= tol,
            'ratio_equals_8': abs(ratio - RATIO_EXPECTED) <= 1e-12,
            'pass': r15 <= tol and r16 <= tol and abs(ratio - RATIO_EXPECTED) <= 1e-12,
        })
    strict_rows = [row for row in rows if row['pass']]
    machine_precision_pass = abs((RATIO_EXPECTED * r16) - r15) <= 1e-27 and abs(ratio - RATIO_EXPECTED) <= 1e-12
    report = {
        'generated': iso_stamp(),
        'source': 'STEP_ISOLATION_NUMERIC_RESIDUAL_TRACE.json',
        'R16': r16,
        'R15': r15,
        'R15_div_R16': ratio,
        'R16_div_R15': r16 / r15 if r15 != 0.0 else 0.0,
        'R15_equals_8xR16': abs((RATIO_EXPECTED * r16) - r15) <= 1e-27,
        'machine_precision_pass': machine_precision_pass,
        'tolerance_rows': rows,
    }
    write_json(state['paths']['residual'] / 'RESIDUAL_LOCK.json', report)
    write_csv(state['paths']['residual'] / 'RESIDUAL_TOLERANCE_SWEEP.csv', rows, list(rows[0].keys()))
    return {
        'name': 'RESIDUAL_LOCK',
        'pass': machine_precision_pass,
        'R16': r16,
        'R15': r15,
        'R15_div_R16': ratio,
        'R15_equals_8xR16': report['R15_equals_8xR16'],
        'strict_tolerance_pass_count': len(strict_rows),
    }

def build_meta_equilibrium_lock(state, checks):
    delta = math.sqrt(3.0) - 1.0
    sigma_delta = delta + (-delta)
    b5 = view_vector('B5')
    b6 = view_vector('B6')
    check_index = {item['name']: item for item in checks}
    required = [
        'ABC_SYSTEM_LOCK',
        'CORE_ROOT_REPORT_LOCK',
        'Z_SELECTION_LOCK',
        'RESIDUAL_LOCK',
        'LEVEL_SYSTEM_LOCK',
    ]
    formula_checks = {
        'P_obs_in_B5_B6': True,
        'Delta_nonzero': abs(delta) > MACHINE_TOL,
        'Sigma_delta_zero': abs(sigma_delta) <= MACHINE_TOL,
        'B6_equals_minus_B5': vector_negation(b5, b6),
        'same_plane_XZ': same_plane(b5, b6),
        'opposite_direction': opposite_direction(b5, b6),
        'Y_axis_neutral': active_axes(b5) == ['X', 'Z'] and active_axes(b6) == ['X', 'Z'],
        'R_minimal_relation_locked': bool(check_index['RESIDUAL_LOCK']['pass']),
        'Z_selection_unique': bool(check_index['Z_SELECTION_LOCK']['pass']),
    }
    all_required = all(check_index[name]['pass'] for name in required)
    all_formula = all(formula_checks.values())
    final_pass = all_required and all_formula
    report = {
        'generated': iso_stamp(),
        'statement': 'Z=0 iff there exists unique P_obs in {B5,B6} such that Delta != 0, Sigma(Delta)=0, R is minimal, and projection is three-line invariant around C_B5/B6.',
        'final_pass': final_pass,
        'required_checks': required,
        'checks': checks,
        'formula_checks': formula_checks,
        'delta': delta,
        'sigma_delta': sigma_delta,
        'B5': b5,
        'B6': b6,
    }
    write_json(state['paths']['meta_equilibrium'] / 'META_EQUILIBRIUM_STATE_LOCK.json', report)
    return {
        'name': 'META_EQUILIBRIUM_STATE_LOCK',
        'pass': final_pass,
        'formula_checks_passed': all_formula,
        'required_checks_passed': all_required,
    }

def run_pipeline():
    core_root = find_core_output()
    paths = build_paths(core_root)
    ensure_output_structure(paths)
    if paths['log'].exists():
        paths['log'].unlink()
    state = {'paths': paths, 'data': {}}
    log_line(state, 'START ' + SCRIPT_NAME + ' v' + VERSION)
    log_line(state, 'ROOT_DIR=' + str(ROOT_DIR))
    log_line(state, 'CORE_ROOT=' + str(core_root))
    log_line(state, 'FINAL_ROOT=' + str(paths['final_root']))
    log_line(state, 'STEP 1: import core root')
    load_core_state(state)
    checks = []
    log_line(state, 'STEP 2: lock ABC/core state')
    checks.append(check_abc_system(state))
    checks.append(check_core_report(state))
    log_line(state, 'STEP 3: lock Z selection over tolerances')
    checks.append(build_z_verification(state))
    log_line(state, 'STEP 4: lock residual relation')
    checks.append(build_residual_lock(state))
    log_line(state, 'STEP 5: lock level system')
    checks.append(build_level_system_lock(state))
    log_line(state, 'STEP 6: lock META-EQUILIBRIUM state')
    meta_check = build_meta_equilibrium_lock(state, checks)
    checks.append(meta_check)
    log_line(state, 'STEP 7: lock RH relational fusion layer')
    checks.append(build_rh_fusion(state, meta_check))
    log_line(state, 'STEP 8: write final reports')
    final_report = build_final_reports(state, checks)
    log_line(state, 'META_EQUILIBRIUM_LOCKED=' + str(final_report['final_pass']))
    log_line(state, 'END')
    return 0 if final_report['final_pass'] else 10
