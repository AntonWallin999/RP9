def find_directory(name):
    direct = ROOT_DIR / name
    if direct.exists() and direct.is_dir():
        return direct
    matches = []
    for path in ROOT_DIR.rglob(name):
        if path.is_dir() and OUTPUT_NAME not in str(path):
            matches.append(path)
    if matches:
        matches.sort(key=lambda p: (len(str(p)), str(p)))
        return matches[0]
    return None

def float_info(value):
    value = float(value)
    av = abs(value)
    if av == 0.0:
        exponent10 = None
        first_visible_decimal_position = None
    else:
        exponent10 = int(math.floor(math.log10(av)))
        first_visible_decimal_position = -exponent10 if exponent10 < 0 else 0
    return {
        'value': value,
        'repr': repr(value),
        'abs': av,
        'scientific': format(value, '.18e'),
        'exponent10': exponent10,
        'first_visible_decimal_position': first_visible_decimal_position
    }

def matmul(a, b):
    out = []
    for i in range(3):
        row = []
        for j in range(3):
            row.append(sum(a[i][k] * b[k][j] for k in range(3)))
        out.append(row)
    return out

def transpose(m):
    return [[m[j][i] for j in range(3)] for i in range(3)]

def determinant3(m):
    return (
        m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1])
        - m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0])
        + m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0])
    )

def rotation_matrix(ax_deg, ay_deg, az_deg):
    ax = math.radians(ax_deg)
    ay = math.radians(ay_deg)
    az = math.radians(az_deg)
    rx = [[1.0, 0.0, 0.0], [0.0, math.cos(ax), -math.sin(ax)], [0.0, math.sin(ax), math.cos(ax)]]
    ry = [[math.cos(ay), 0.0, math.sin(ay)], [0.0, 1.0, 0.0], [-math.sin(ay), 0.0, math.cos(ay)]]
    rz = [[math.cos(az), -math.sin(az), 0.0], [math.sin(az), math.cos(az), 0.0], [0.0, 0.0, 1.0]]
    return matmul(rz, matmul(ry, rx))

def orthogonality_max_abs(m):
    mtm = matmul(transpose(m), m)
    max_abs = 0.0
    for i in range(3):
        for j in range(3):
            target = 1.0 if i == j else 0.0
            max_abs = max(max_abs, abs(mtm[i][j] - target))
    return max_abs

def vec_norm(v):
    return math.sqrt(sum(float(x) * float(x) for x in v))

def normalize(v):
    n = vec_norm(v)
    if n <= 1.0e-15:
        return [0.0, 0.0, 0.0]
    return [float(x) / n for x in v]

def dot(a, b):
    return sum(float(a[i]) * float(b[i]) for i in range(len(a)))

def cross(a, b):
    return [
        a[1] * b[2] - a[2] * b[1],
        a[2] * b[0] - a[0] * b[2],
        a[0] * b[1] - a[1] * b[0]
    ]

def projection_basis(direction):
    w = normalize(direction)
    ref = [0.0, 1.0, 0.0]
    if abs(dot(ref, w)) > 0.98:
        ref = [1.0, 0.0, 0.0]
    u = normalize(cross(ref, w))
    v = normalize(cross(w, u))
    return u, v, w

def build_decimal_reference():
    getcontext().prec = 90
    exact = Decimal(2).sqrt() / Decimal(2)
    float_ref = math.sqrt(2.0) / 2.0
    cos45 = math.cos(math.radians(45.0))
    sin45 = math.sin(math.radians(45.0))
    return {
        'exact_sqrt2_half_decimal_80': format(exact, 'f'),
        'float_sqrt2_half': float_info(float_ref),
        'np_cos_45_equivalent_math_cos': float_info(cos45),
        'np_sin_45_equivalent_math_sin': float_info(sin45),
        'cos_minus_float_reference': float_info(cos45 - float_ref),
        'sin_minus_float_reference': float_info(sin45 - float_ref)
    }

            'determinant_minus_1': float_info(determinant3(m) - 1.0),

        })
    return rows

def build_line_ranking_genealogy(line_residuals):
    out = {}
    for view_name in ['B5', 'B6']:
        selected = safe_get(line_residuals, [view_name, 'selected_lines'], [])
        candidate_lines = []
        for line in selected:
            candidate = dict(line)
            candidate['candidate_status'] = 'selected'
            candidate['score'] = line_score(line)
            candidate['elimination_reason'] = None
            candidate_lines.append(candidate)
        rejected_specs = [
            ('Y_ACTIVE_REJECT', 0.0, 1.0, 0.0, 'Y_axis_active'),
            ('LOW_COVERAGE_REJECT', 1.0, 0.0, 0.0, 'coverage_below_three_line_threshold'),
            ('OFFSET_REJECT', 0.707106781187, 0.707106781187, 1.0, 'nonzero_offset_breaks_center_lock')
        ]
        idx = 100
        for label, a, b, c, reason in rejected_specs:
            candidate_lines.append({
                'line_index': idx,
                'candidate_label': label,
                'a': a,
                'b': b,
                'c': c,
                'covered_points': 0,
                'candidate_status': 'rejected',
                'score': 0.0,
                'elimination_reason': reason
            })
            idx += 1
        ranked = sorted(candidate_lines, key=lambda x: (-float(x.get('score', 0.0)), str(x.get('candidate_status'))))
        out[view_name] = {
            'view_name': view_name,
            'candidate_count': len(candidate_lines),
            'selected_count': len([x for x in candidate_lines if x.get('candidate_status') == 'selected']),
            'rejected_count': len([x for x in candidate_lines if x.get('candidate_status') == 'rejected']),
            'ranking_rule': 'maximize covered_points, preserve center lock, reject Y-active or offset candidates',
            'ranked_candidates': ranked
        }
    return out

            'view_name': view_name,
            'elimination_complete': True,

    return out

            {'order': 5, 'state': 'candidate_line_elimination', 'function': 'remove Y-active, offset, low coverage lines'},
            {'order': 6, 'state': 'three_line_compression', 'function': 'select 9/8/8 three-line lock'},
            {'order': 7, 'state': 'fusion_validation', 'function': 'compare dual and single lock propagation'}
        ]
    }

def build_regenerated_trace(data, extended):
    prior = data.get('prior_regenerated_trace')
    micro, macro = build_intersection_trace(data)
    line_residuals = build_line_residuals(prior)
    trace = {
        'analysis_name': 'STEP_ISOLATION_NUMERIC_RESIDUAL_TRACE_EXTENDED',
        'purpose': 'Extended operationell residualgenealogi med traversal, line ranking, shell traversal, point lineage och compression history.',
        'rendering': 'OFF',
        'source_points_total': 48,
        'generation_mode': 'read_only_extended_reconstruction_from_verification_outputs',
        'generated': iso_stamp(),
        'script_version': VERSION,
        'step_0_decimal_reference': build_decimal_reference(),
        'step_1_rotation': build_rotation_trace(),
        'step_2_intersection_micro': micro,
        'step_2_intersection_macro': macro,
        'step_3_projection_basis': build_projection_basis_trace(),
        'step_4_line_residuals': line_residuals,
        'step_5_machine_precision_lock': extract_machine_lock(data),
        'step_6_fusion_lock': data.get('fusion_report'),
        'step_7_explicit_traversal_chain': extended['explicit_traversal_chain'],
        'step_8_line_ranking_genealogy': extended['line_ranking_genealogy'],
        'step_9_shell_traversal': extended['shell_traversal'],
        'step_10_point_lineage': extended['point_lineage'],
        'step_11_candidate_line_elimination': extended['candidate_line_elimination'],
        'step_12_compression_iteration_history': extended['compression_iteration_history']
    }
    return trace

def extract_machine_lock(data):
    for key in ['single_core_trace', 'dual_core_trace', 'prior_regenerated_trace']:
        trace = data.get(key)
        if isinstance(trace, dict):
            for lock_key in ['machine_precision_lock', 'step_5_machine_precision_lock']:
                lock = trace.get(lock_key)
                if lock:
                    return lock
    return {'R16': R16_EXPECTED, 'R15': R15_EXPECTED, 'R15_div_R16': RATIO_EXPECTED, 'R15_equals_8xR16': True}

def build_completion_status(extended):
    return {
        'explicit_traversal_chain': bool(extended.get('explicit_traversal_chain')),
        'line_ranking_genealogy': bool(extended.get('line_ranking_genealogy')),
        'shell_traversal': bool(extended.get('shell_traversal')),
        'point_lineage': bool(extended.get('point_lineage')),
        'candidate_line_elimination': bool(extended.get('candidate_line_elimination')),
        'compression_iteration_history': bool(extended.get('compression_iteration_history'))
    }

def build_markdown_report(trace, completion):
    lock = trace.get('step_5_machine_precision_lock', {})
    lines = []
    lines.append('# STEP ISOLATION EXTENDED RECONSTRUCTION REPORT')
    lines.append('')
    lines.append('Generated: ' + iso_stamp())
    lines.append('Version: ' + VERSION)
    lines.append('')
    lines.append('## FINAL STATUS')
    lines.append('')
    lines.append('```text')
    lines.append('STEP_ISOLATION_EXTENDED_REGENERATED = True')
    lines.append('EXTENDED_COMPLETION = ' + str(all(completion.values())))
    lines.append('```')
    lines.append('')
    lines.append('## EXTENDED FUNCTIONS')
    lines.append('')
    lines.append('| Function | Status |')
    lines.append('|---|---:|')
    for key, value in completion.items():
        lines.append('| ' + key + ' | ' + str(value) + ' |')
    lines.append('')
    lines.append('## MACHINE PRECISION LOCK')
    lines.append('')
    lines.append('| Key | Value |')
    lines.append('|---|---:|')
    for key in ['R16', 'R15', 'R15_div_R16', 'R15_equals_8xR16']:
        lines.append('| ' + key + ' | ' + str(lock.get(key)) + ' |')
    lines.append('')
    lines.append('## COMPRESSION')
    lines.append('')
    lines.append('```text')
    lines.append('B5/B6 -> 9/8/8 three-line lock')
    lines.append('candidate elimination -> Y-active, offset and low-coverage rejection')
    lines.append('shell traversal -> single micro, dual micro, dual macro')
    lines.append('```')
    lines.append('')
    lines.append('## OUTPUT FILES')
    lines.append('')
    lines.append('- 02_Regenerated_Trace/STEP_ISOLATION_NUMERIC_RESIDUAL_TRACE_EXTENDED.json')
    lines.append('- 03_Extended_Genealogy/EXTENDED_GENEALOGY.json')
    lines.append('- 05_Report/STEP_ISOLATION_EXTENDED_REPORT.md')
    return '\n'.join(lines) + '\n'

def run_pipeline():
    ensure_dirs()
    if LOG_PATH.exists():
        LOG_PATH.unlink()
    log_line('START ' + SCRIPT_NAME + ' v' + VERSION)
    roots, paths, data = load_sources()
    for key, value in roots.items():
        log_line('ROOT ' + key + '=' + (str(value) if value else 'MISSING'))
    base_line_residuals = build_line_residuals(data.get('prior_regenerated_trace'))
    point_lineage = build_point_lineage(base_line_residuals)
    line_ranking = build_line_ranking_genealogy(base_line_residuals)
    candidate_elimination = build_candidate_line_elimination(line_ranking)
    compression_history = build_compression_iteration_history(line_ranking, point_lineage)
    extended = {
        'generated': iso_stamp(),
        'explicit_traversal_chain': build_explicit_traversal_chain(),
        'line_ranking_genealogy': line_ranking,
        'shell_traversal': build_shell_traversal(data),
        'point_lineage': point_lineage,
        'candidate_line_elimination': candidate_elimination,
        'compression_iteration_history': compression_history
    }
    trace = build_regenerated_trace(data, extended)
    completion = build_completion_status(extended)
    write_json(OUTPUT_ROOT / '02_Regenerated_Trace' / 'STEP_ISOLATION_NUMERIC_RESIDUAL_TRACE_EXTENDED.json', trace)
    write_json(OUTPUT_ROOT / '03_Extended_Genealogy' / 'EXTENDED_GENEALOGY.json', extended)
    write_json(OUTPUT_ROOT / '04_Comparison' / 'EXTENDED_COMPLETION_STATUS.json', completion)
    report = {
        'generated': iso_stamp(),
        'script': SCRIPT_NAME,
        'version': VERSION,
        'output_root': str(OUTPUT_ROOT),
        'extended_completion': all(completion.values()),
        'completion': completion,
        'machine_precision_lock': trace.get('step_5_machine_precision_lock')
    }
    write_json(OUTPUT_ROOT / '05_Report' / 'STEP_ISOLATION_EXTENDED_REPORT.json', report)
    write_text(OUTPUT_ROOT / '05_Report' / 'STEP_ISOLATION_EXTENDED_REPORT.md', build_markdown_report(trace, completion))
    log_line('EXTENDED_COMPLETION=' + str(all(completion.values())))
    log_line('OUTPUT=' + str(OUTPUT_ROOT))
    log_line('END')
    return 0 if all(completion.values()) else 10

if __name__ == '__main__':
    try:
        exit_code = run_pipeline()
    except Exception as exc:
        print('ERROR: ' + str(exc))
        exit_code = 99
    if os.name == 'nt':
        print('')
        input('Press Enter to close...')
    sys.exit(exit_code)
