RP9. CORE4. (M-P-V). PYTHON


RP9_CORE4_MANIFEST_POSITION_VALIDATOR.py
Version: 1.0.0

Syfte:
    Besvarar den faktiska frågan:

    När Vesica Piscis ligger rakt fram i XY-planet och H-D-R-sekvensen
    genererar Core4, vilken observationsposition / orienteringslinje
    manifesteras av operatorn själv?

    Sammanfaller den operatoriskt härledda Core4-positionen med B5-N6 / B6-N6,
    eller krävs ett efterföljande observationsval?

Princip:
    Koden börjar inte med B5/B6.
    Koden härleder först positionen från startläget:

        Vesica Piscis frontläge = XY-plan
        relationell längd = X
        frontal observationsnormal = Z
        H-D-R halva-vridning mellan 0° och 90° = 45° i XZ-relationen

    Först efter att den positionen är härledd jämförs den med ABC/B5/B6.

Output:
    - JSON-rapport
    - CSV med axel-/positionsjämförelser
    - Markdownrapport
    - Logg

Exit code:
    0  = testet kunde besvara frågan och positionen matchar B5/B6-linjen
    10 = testet kunde besvara frågan men positionen matchar inte B5/B6-linjen
    20 = testet är definitionsmässigt otillräckligt

def ensure_dirs() -> None:

def now() -> str:
    return datetime.now().isoformat(timespec="seconds")


def log(message: str) -> None:
    ensure_dirs()

def write_text(path: Path, text: str) -> None:

def write_json(path: Path, payload: object) -> None:

def write_csv(path: Path, rows: List[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    if not rows:
        path.write_text("", encoding="utf-8")
        return
    with path.open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(rows[0].keys()))
        writer.writeheader()
        writer.writerows(rows)

def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def collect_reference_files() -> List[ReferenceFileRecord]:
    records: List[ReferenceFileRecord] = []
    for name in REFERENCE_FILES:
        path = ROOT_DIR / name
        if path.exists():
            records.append(ReferenceFileRecord(name, True, sha256_file(path), path.stat().st_size))
        else:
            records.append(ReferenceFileRecord(name, False, None, None))
    return records


def parse_assignment_from_python(path: Path, name: str):
    if not path.exists():
        return None
    try:
        tree = ast.parse(path.read_text(encoding="utf-8"))
    except Exception:
        return None
    for node in tree.body:
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name) and target.id == name:
                    try:
                        return ast.literal_eval(node.value)
                    except Exception:
                        return None
    return None

    source = ROOT_DIR / "Vesica_Render_HTML_Only_AB_Categorized.py"
    rotations = parse_assignment_from_python(source, "ROTATIONS_GREEN")
    view_groups = parse_assignment_from_python(source, "VIEW_GROUPS")
    info = {
        "rotations_source": "fallback",
        "view_groups_source": "fallback",
    }
    if rotations is not None:
        rotations = [tuple(float(x) for x in rot) for rot in rotations]
        info["rotations_source"] = str(source.name)

    if view_groups is not None:
        info["view_groups_source"] = str(source.name)
    else:
        view_groups = FALLBACK_VIEW_GROUPS
    return rotations, view_groups, info


# ---------------------------------------------------------------------------
# Vektorgeometri
# ---------------------------------------------------------------------------
def normalize(v: Iterable[float]) -> np.ndarray:
    arr = np.asarray(list(v), dtype=float)
    n = float(np.linalg.norm(arr))
    if n <= EPS:
        raise ValueError(f"Kan inte normalisera nollvektor: {arr}")
    return arr / n


def angle_deg_directed(a: Iterable[float], b: Iterable[float]) -> float:
    ua = normalize(a)
    ub = normalize(b)
    dot = float(np.dot(ua, ub))
    dot = max(-1.0, min(1.0, dot))
    return float(math.degrees(math.acos(dot)))


def angle_deg_line(a: Iterable[float], b: Iterable[float]) -> float:
    ua = normalize(a)
    ub = normalize(b)
    dot = abs(float(np.dot(ua, ub)))
    dot = max(-1.0, min(1.0, dot))
    return float(math.degrees(math.acos(dot)))


def same_directed(a: Iterable[float], b: Iterable[float]) -> bool:
    return angle_deg_directed(a, b) <= ANGLE_TOL_DEG


def same_line(a: Iterable[float], b: Iterable[float]) -> bool:
    return angle_deg_line(a, b) <= ANGLE_TOL_DEG


def rotation_matrix(ax_deg: float, ay_deg: float, az_deg: float) -> np.ndarray:
    ax, ay, az = np.deg2rad([ax_deg, ay_deg, az_deg])
    rx = np.array([
        [1.0, 0.0, 0.0],
        [0.0, np.cos(ax), -np.sin(ax)],
        [0.0, np.sin(ax), np.cos(ax)],
    ])
    ry = np.array([
        [np.cos(ay), 0.0, np.sin(ay)],
        [0.0, 1.0, 0.0],
        [-np.sin(ay), 0.0, np.cos(ay)],
    ])
    rz = np.array([
        [np.cos(az), -np.sin(az), 0.0],
        [np.sin(az), np.cos(az), 0.0],
        [0.0, 0.0, 1.0],
    ])
    return rz @ ry @ rx


def round_vec(v: Iterable[float], decimals: int = 12) -> List[float]:
    arr = np.asarray(list(v), dtype=float)
    arr[np.abs(arr) < 10 ** (-(decimals + 1))] = 0.0
    return [round(float(x), decimals) for x in arr]

def derive_manifest_position_from_vp() -> Dict[str, object]:
    """
    Detta är huvudtestet.

    Startdefinition:
        Vesica Piscis frontläge ligger i XY.
        X = relationell längd / centrumrelation.
        Y = höjd i VP-planet.
        Z = frontal observationsnormal.

    H-D-R-frågan som testas:
        Den första 3D-aktiveringen inför 90° mellan frontnormalen Z
        och den relationella horisontalen X.

        Den andra H-D-R-aktiveringen från VP ger halva vinkeln mellan 0° och 90°.
        Halvvinkeln är 45° och riktningen blir normaliserad X + Z.

    Det ger en riktad position utan att B5/B6 används som input.
    """
    x_relation = normalize([1.0, 0.0, 0.0])
    y_height = normalize([0.0, 1.0, 0.0])
    z_front = normalize([0.0, 0.0, 1.0])

    # Operatorisk halvvinkel i XZ-relationen:
    manifest = normalize(x_relation + z_front)

    # Kontroll: vinklarna mot X och Z skall båda vara 45°.
    angle_to_x = angle_deg_directed(manifest, x_relation)
    angle_to_z = angle_deg_directed(manifest, z_front)
    y_component_zero = abs(float(np.dot(manifest, y_height))) <= EPS

    return {
        "x_relation_axis": round_vec(x_relation),
        "y_height_axis": round_vec(y_height),
        "z_front_normal": round_vec(z_front),
        "manifest_position_vector": round_vec(manifest),
        "angle_to_x_relation_deg": angle_to_x,
        "angle_to_z_front_deg": angle_to_z,
        "y_component_zero": y_component_zero,
        "half_angle_lock_45_45": abs(angle_to_x - 45.0) <= ANGLE_TOL_DEG and abs(angle_to_z - 45.0) <= ANGLE_TOL_DEG,
    }


def flatten_abc_views(view_groups: Dict[str, dict]) -> Dict[str, Tuple[float, float, float]]:
    views: Dict[str, Tuple[float, float, float]] = {}
    for group_key, group_info in view_groups.items():
        for name, vec in group_info.get("views", {}).items():
            views[name] = tuple(float(x) for x in vec)
    return views


def compare_manifest_to_views(manifest_vec: Iterable[float], views: Dict[str, Tuple[float, float, float]]) -> List[OrientationComparison]:
    manifest = normalize(manifest_vec)
    rows: List[OrientationComparison] = []
    for name, vec in views.items():
        v = normalize(vec)
        signed = float(np.dot(manifest, v))
        rows.append(OrientationComparison(
            candidate_name=name,
            vector=round_vec(v),
            signed_dot=signed,
            abs_dot=abs(signed),
            angle_deg_directed=angle_deg_directed(manifest, v),
            angle_deg_line=angle_deg_line(manifest, v),
            is_same_directed_position=same_directed(manifest, v),
            is_same_axis_line=same_line(manifest, v),
        ))
    rows.sort(key=lambda r: (r.angle_deg_line, r.angle_deg_directed, r.candidate_name))
    return rows


def extract_core4_axes(rotations: List[Tuple[float, float, float]], manifest_vec: Iterable[float]) -> List[Core4AxisRecord]:
    base_axes = {
        "relation_axis_X_center_pair": np.array([1.0, 0.0, 0.0], dtype=float),
        "height_axis_Y_circle_plane": np.array([0.0, 1.0, 0.0], dtype=float),
        "normal_axis_Z_circle_plane": np.array([0.0, 0.0, 1.0], dtype=float),
    }
    manifest = normalize(manifest_vec)
    records: List[Core4AxisRecord] = []
    for rot in rotations:
        mat = rotation_matrix(*rot)
        for role, base in base_axes.items():
            vec = normalize(mat @ base)
            records.append(Core4AxisRecord(
                rotation_deg=tuple(float(x) for x in rot),
                axis_role=role,
                vector=round_vec(vec),
                signed_dot_to_manifest=float(np.dot(vec, manifest)),
                angle_deg_to_manifest_line=angle_deg_line(vec, manifest),
                is_manifest_axis=same_line(vec, manifest),
            ))
    return records


def classify(
    derivation: Dict[str, object],
    comparisons: List[OrientationComparison],
    core4_axes: List[Core4AxisRecord],
) -> Dict[str, object]:
    same_directed = [r.candidate_name for r in comparisons if r.is_same_directed_position]
    same_line_names = [r.candidate_name for r in comparisons if r.is_same_axis_line]
    core4_manifest_axes = [r for r in core4_axes if r.is_manifest_axis]

    b5_directed = "B5" in same_directed
    b6_directed = "B6" in same_directed
    b5b6_line = "B5" in same_line_names and "B6" in same_line_names
    manifest_in_core4 = len(core4_manifest_axes) > 0
    half_angle_lock = bool(derivation["half_angle_lock_45_45"])
    y_zero = bool(derivation["y_component_zero"])

    if b5_directed and b5b6_line and manifest_in_core4 and half_angle_lock and y_zero:
        classification = "CORE4_MANIFEST_POSITION_IS_B5_WITH_B6_AS_INVERSE_LINE"
        answer = (
            "Core4:s manifesterade utgångsposition är B5-linjen: normaliserad riktning (1, 0, 1). "
            "B6 är samma observationslinje i inverterad riktning (-1, 0, -1). "
            "Positionen är därmed operatoriskt genererad från H-D-R-halvvinkeln mellan X och Z, inte vald i efterhand."
        )
        final_pass = True
        exit_code = 0
    elif b5b6_line and manifest_in_core4 and half_angle_lock and y_zero:
        classification = "CORE4_MANIFEST_POSITION_IS_B5B6_LINE_DIRECTION_UNRESOLVED"
        answer = (
            "Core4:s manifesterade utgångslinje sammanfaller med B5/B6-axeln, "
            "men riktningen B5 kontra B6 är inte låst av de givna definitionsvillkoren."
        )
        final_pass = True
        exit_code = 0
    elif half_angle_lock and y_zero and manifest_in_core4:
        classification = "CORE4_MANIFEST_POSITION_GENERATED_BUT_NOT_B5B6"
        answer = (
            "Core4 genererar en 45°-position i XZ-fältet, men den matchar inte B5/B6 enligt ABC-definitionerna."
        )
        final_pass = False
        exit_code = 10
    else:
        classification = "INSUFFICIENT_DEFINITION_OR_FAILED_DERIVATION"
        answer = (
            "Validatorn kan inte låsa frågan eftersom halvvinkeln, Y=0-villkoret eller Core4-matchningen fallerade."
        )
        final_pass = False
        exit_code = 20

    nearest = comparisons[0] if comparisons else None
    return {
        "final_pass": final_pass,
        "exit_code": exit_code,
        "classification": classification,
        "answer": answer,
        "directed_position_matches": same_directed,
        "axis_line_matches": same_line_names,
        "nearest_abc_view": nearest.candidate_name if nearest else None,
        "nearest_abc_line_angle_deg": nearest.angle_deg_line if nearest else None,
        "manifest_position_in_core4_axes": manifest_in_core4,
        "core4_manifest_axis_count": len(core4_manifest_axes),
        "half_angle_lock_45_45": half_angle_lock,
        "y_component_zero": y_zero,
        "b5_directed_match": b5_directed,
        "b6_directed_match": b6_directed,
        "b5b6_axis_line_match": b5b6_line,
    }

def zip_outputs() -> None:
    if ZIP_PATH.exists():
        ZIP_PATH.unlink()
    files = [Path(__file__).resolve()]
    if OUT_DIR.exists():
        files.extend([p for p in OUT_DIR.rglob("*") if p.is_file()])
    with zipfile.ZipFile(ZIP_PATH, "w", compression=zipfile.ZIP_DEFLATED) as zf:
        for p in files:
            zf.write(p, arcname=p.relative_to(ROOT_DIR))


def run() -> int:
    ensure_dirs()
    if LOG_PATH.exists():
        LOG_PATH.unlink()
    log("START CORE4 MANIFEST POSITION VALIDATOR")

    rotations, view_groups, source_info = load_rotations_and_views()
    references = collect_reference_files()

    derivation = derive_manifest_position_from_vp()
    manifest = derivation["manifest_position_vector"]
    abc_views = flatten_abc_views(view_groups)
    comparisons = compare_manifest_to_views(manifest, abc_views)
    core4_axes = extract_core4_axes(rotations, manifest)
    conclusion = classify(derivation, comparisons, core4_axes)

    payload = {
        "metadata": {
            "generated": now(),
            "script": SCRIPT_NAME,
            "version": SCRIPT_VERSION,
            "validator_type": "manifest_position_derivation_not_candidate_uniqueness",
            "angle_tolerance_deg": ANGLE_TOL_DEG,
            "source_info": source_info,
        },
        "hypothesis": {
            "question": "Är Core4 positionerad enligt B5-N6/B6-N6 från början, eller är positionen vald efter etablering?",
            "H1_operator_generated": "Core4:s manifesterade utgångsposition härleds från H-D-R-halvvinkeln och sammanfaller med B5/B6-linjen.",
            "H0_selected_after_establishment": "Core4 etableras utan B5/B6-position; B5/B6 måste väljas efteråt som observationsposition.",
        },
        "reference_files": [asdict(r) for r in references],
        "operator_derivation": derivation,
        "abc_position_comparison": [asdict(r) for r in comparisons],
        "core4_axis_comparison": [asdict(r) for r in core4_axes],
        "conclusion": conclusion,
    }

    write_json(JSON_DIR / "CORE4_MANIFEST_POSITION_VALIDATION.json", payload)
    write_csv(CSV_DIR / "ABC_POSITION_COMPARISON.csv", [asdict(r) for r in comparisons])
    write_csv(CSV_DIR / "CORE4_AXIS_COMPARISON.csv", [asdict(r) for r in core4_axes])
    write_text(REPORT_DIR / "CORE4_MANIFEST_POSITION_VALIDATION.md", build_report(payload))

    log(f"FINAL_PASS={conclusion['final_pass']}")
    log(f"CLASSIFICATION={conclusion['classification']}")
    log(f"DIRECTED_POSITION_MATCHES={conclusion['directed_position_matches']}")
    log(f"AXIS_LINE_MATCHES={conclusion['axis_line_matches']}")
    log(f"ANSWER={conclusion['answer']}")

    zip_outputs()
    log(f"ZIP={ZIP_PATH}")
    log("END")
    return int(conclusion["exit_code"])


if __name__ == "__main__":
    sys.exit(run())
